
wsv_great_war_decisions = {

	### --------------- Salonika Front ----------------- ###

	## dummy decision which when howevered over show # of divisions and men you've committed to Macedonia
	macedonia_commitment_overview = {
		icon = generic_operation
		priority = 10

		allowed = {
			always = yes
		} 
		available = {
			always = no
		}
		visible = {
			has_global_flag = allied_army_orient_formed
			custom_trigger_tooltip = {
				tooltip = has_troops_in_salonika_tt
				check_variable = { ROOT.num_orient_divisions > 0 }
			}
		}

		state_target = yes
		on_map_mode = decision_view_only
		target_trigger  = { 
			# FROM = {
			# 	is_owned_by = var:global.ARMY_ORIENT_ACTUAL
			# }
			FROM = {
				state = 855
			}
		}

		cost = 0
		days_remove = 0

		complete_effect = {}
	}


	# Fix supply lines in Greece cause AI wont
	build_ports_in_salonika = {
		icon = generic_construction

		allowed = {
			always = yes
		} 
		available = {
			always = yes
			var:global.ARMY_ORIENT_ACTUAL = {
				controls_state = 855   # thessalonkia
			}
		}
		visible = {
			has_global_flag = allied_army_orient_formed
			is_major = yes
			tag = var:global.ARMY_ORIENT_OVERLORD
		}

		fire_only_once = yes
		days_remove = 60

		custom_cost_trigger = {
			num_of_civilian_factories > 3
			command_power > 25
		}
		custom_cost_text = decision_cost_build_ports_in_salonika

		modifier = {
			civilian_factory_use = 3
		}

		complete_effect = {
			add_command_power = -25
		}

		remove_effect = {
			855 = {
				add_building_construction = {
					type = naval_base
					level = 7  # does it need to be beefier? idk lol
					province = 11818
				}
			}
		}

		ai_will_do = {
			factor = 100
		}
	}


	# Send troops to Salonika front
	bolster_salonika_front = {
		icon = generic_military

		allowed = {
			always = yes
		} 
		available = {
			var:global.ARMY_ORIENT_ACTUAL = {
				controls_state = 855   # thessalonkia
			}
			custom_trigger_tooltip = {
				tooltip = max_orient_divisions_tt   ## TODO - loc is broken
				set_temp_variable = { ROOT.max_division_to_orient = ROOT.num_armies }
				multiply_temp_variable = { ROOT.max_division_to_orient = 0.15 }  	# max 15% of army
				check_variable = { ROOT.max_division_to_orient < ROOT.num_orient_divisions }
			}
		}
		visible = {
			has_global_flag = allied_army_orient_formed
			is_major = yes
			OR = {
				tag = var:global.ARMY_ORIENT_OVERLORD
				is_in_faction_with = var:global.ARMY_ORIENT_OVERLORD
				capital_scope = {
					is_on_continent = europe
				}
			}
			OR = {
				has_war_with = BUL
				has_war_with = AUH
			}
		}

		days_remove = 60
		days_re_enable = 0

		custom_cost_trigger = {
			has_manpower > 41000
			command_power > 10
		}
		custom_cost_text = decision_cost_bolster_salonika_front

		complete_effect = {
			add_command_power = -10
			ROOT = { send_3_divisions_to_army_orient = yes }
		}

		remove_effect = {
			var:global.ARMY_ORIENT_ACTUAL = { spawn_3_orient_infantry_divisions = yes }
			hidden_effect = {
				if = {
					limit = { NOT = { tag = var:global.ARMY_ORIENT_OVERLORD } }
					var:global.ARMY_ORIENT_OVERLORD = { country_event = wsv_armyorient.2 }
				}
			}
		}

		ai_will_do = {
			factor = 0 
			modifier = {
				add = 20
				OR = {
					tag = FRA
					tag = ENG
				}
				var:global.ARMY_ORIENT_ACTUAL = {
					has_army_manpower = { size < 399999 }
				}
			}
		}
	}

	# activates decision to 'open the Macedonia front'  (ROOT is ARMY_ORIENT_OVERLORD)
	open_macedonia_front_theatre = {
		icon = generic_ignite_civil_war

		allowed = {
			always = yes
		} 
		available = {
			tag = var:global.ARMY_ORIENT_OVERLORD
			var:global.ARMY_ORIENT_ACTUAL = {
				controls_state = 855   # thessalonkia
				OR = {
					has_army_manpower = { size > 399999 }
					AND = {
						SER = { controls_state =  759 }
						has_army_manpower = { size > 149999 }
					}
				}
			}
		}
		visible = {  # everyone can see it, only overlord can trigger it
			has_global_flag = allied_army_orient_formed
			is_major = yes
			OR = {
				tag = var:global.ARMY_ORIENT_OVERLORD
				is_in_faction_with = var:global.ARMY_ORIENT_OVERLORD
				capital_scope = {
					is_on_continent = europe
				}
			}
			OR = {
				has_war_with = BUL
				has_war_with = AUH
			}
		}

		custom_cost_trigger = {
			command_power > 25
		}
		custom_cost_text = decision_cost_open_macedonia_front_theatre

		fire_only_once = yes
		days_remove = 30

		complete_effect = {
			add_command_power = -25
			custom_effect_tooltip = macedonia_front_premove_tt
			hidden_effect = {
				every_state = {
					limit = {
						OR = { state=184  state=731  state=867  state=758  state=803 }
						is_owned_by = GRE
					}
					transfer_state_to = var:global.ARMY_ORIENT_ACTUAL
				}
				BUL = {
					if = {
						limit = { has_war_with = ROOT }
						transfer_state = 184
						country_event = wsv_armyorient.4
					}
				}
				AUH = {
					if = {
						limit = { has_war_with = ROOT }
					}
					country_event = wsv_armyorient.4
				}
			}
		}

		war_with_on_remove = AUH
		war_with_on_remove = BUL

		remove_effect = {  
			# add to faction
			puppet = var:global.ARMY_ORIENT_ACTUAL
			set_autonomy = {
				target = var:global.ARMY_ORIENT_ACTUAL
				autonomous_state = autonomy_puppet
			}
			add_to_faction = var:global.ARMY_ORIENT_ACTUAL
			# add to wars
			if = {
				limit = { has_war_with = BUL }
				var:global.ARMY_ORIENT_ACTUAL = {
					add_to_war = {
						targeted_alliance = ROOT
						enemy = BUL
					}
				}
			}
			if = {
				limit = { has_war_with = AUH }
				var:global.ARMY_ORIENT_ACTUAL = {
					add_to_war = {
						targeted_alliance = ROOT
						enemy = AUH
					}
				}
			}
			set_global_flag = macedonia_front_opened
		}

		ai_will_do = {
			factor = 100
		}
	}
}
