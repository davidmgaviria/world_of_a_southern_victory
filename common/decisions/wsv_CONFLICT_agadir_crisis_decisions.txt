

##################################################
###########  AGADIR CRISIS DECISIONS ##########

agadir_crisis_cat = {

	# time limit for intervening mission (on timeout Germany fails)
	agadir_relevance_of_morocco_occupation = {
		priority = 25
		icon = generic_ignite_civil_war

		visible = {
			OR = {
				tag = GER
				tag = FRA
				is_in_faction_with = GER
				is_in_faction_with = FRA
				AND = {
					tag = ENG
					has_global_flag = britain_backs_france_in_agadir
				}
			}
		}
		activation = {
			has_global_flag = france_occupies_morocco
			NOT = { has_global_flag = agadir_crisis_over }
		}
		available = {
			always = no
		}
		cancel_trigger = {   # conditions that 'terminate' the mission
			OR = {
				has_global_flag = agadir_crisis_over 
				FRA = { has_war_with = GER }
				FRA = { exists = no }
			} 
		}

		days_mission_timeout = 150
		fire_only_once = yes
		is_good = no

		timeout_effect = {
			set_global_flag = agadir_crisis_over
			if = {
				limit = { tag = GER }
				country_event = wsv_agadir.27
			}
		}
	}


	## Germany sends demands to France
	agadir_germany_send_gunboat = {
		priority = 20
		icon = generic_naval

		visible = {
			has_active_mission = agadir_relevance_of_morocco_occupation
			tag = GER
			NOT = { has_country_flag = sent_gunboat_to_agadir }
		}
		available = {
			has_active_mission = agadir_relevance_of_morocco_occupation
			has_completed_focus = GER_challenge_anglo_french_alliance
			has_country_flag = time_for_action_in_agadir  # France has ignored our inital request for compensation
			has_war = no
		}

		fire_only_once = yes
		days_remove = 4
		cost = 30

		remove_effect = {
			news_event = wsv_agadir.7
		}

		ai_will_do = {
			factor = 50
		}
	}

	### ------------------ Negotiations ------------------ ####
	
	# Inital harsh German demands
	agadir_germany_demands_compensation = {
		icon = generic_operation
		priority = 15

		visible = {
			has_active_mission = agadir_relevance_of_morocco_occupation
			has_global_flag = agadir_crisis
			tag = GER
		}
		available = {
			has_global_flag = britain_response_to_agadir
		}

		fire_only_once = yes
		days_remove = 7
		cost = 0

		complete_effect = {
			set_global_flag = agadir_negotiations_ongoing
			effect_tooltip = {
				add_days_mission_timeout = {
					mission = agadir_relevance_of_morocco_occupation
					days = 30
				}
			}
			hidden_effect = {
				every_country = {
					limit = { has_active_mission = agadir_relevance_of_morocco_occupation }
					add_days_mission_timeout = {
						mission = agadir_relevance_of_morocco_occupation
						days = 30
					}
				}
			}
		}

		remove_effect = {
			FRA = { country_event = wsv_agadir.12 }
			custom_effect_tooltip = agadir_negoitations_begin_tt
		}

		ai_will_do = {
			factor = 50
		}
	}


	# Inital harsh German demands
	agadir_germany_retry_demands_compensation = {
		icon = generic_operation
		priority = 12

		visible = {
			has_active_mission = agadir_relevance_of_morocco_occupation
			has_global_flag = agadir_crisis
			tag = GER
			has_global_flag = agadir_france_rejected_inital_german_demands
		}
		available = {
			always = yes
		}

		fire_only_once = yes
		days_remove = 7
		cost = 25

		complete_effect = {
			set_global_flag = agadir_negotiations_ongoing
			effect_tooltip = {
				add_days_mission_timeout = {
					mission = agadir_relevance_of_morocco_occupation
					days = 30
				}
			}
			hidden_effect = {
				every_country = {
					limit = { has_active_mission = agadir_relevance_of_morocco_occupation }
					add_days_mission_timeout = {
						mission = agadir_relevance_of_morocco_occupation
						days = 30
					}
				}
			}
		}

		remove_effect = {
			FRA = { country_event = wsv_agadir.16 }   # delay in sending next one
		}

		ai_will_do = {
			factor = 50
		}
	}




# 	### ------------------ Escalation ------------------ ####

	# Germany escalates by demanding withdrawal
	agadir_germany_escalates_crisis = {
		icon = generic_prepare_civil_war
		priority = 10

		visible = {
			has_active_mission = agadir_relevance_of_morocco_occupation
			has_global_flag = agadir_crisis
			tag = GER
		}
		available = {
			has_global_flag = britain_response_to_agadir
			NOT = { has_global_flag = agadir_negotiations_ongoing }
		}

		fire_only_once = yes
		days_remove = 4
		cost = 75

		remove_effect = {
			news_event = wsv_agadir.22
			custom_effect_tooltip = likely_war_with_france_tt
		}

		ai_will_do = {
			factor = 0
			# unhistorical modifiers
			modifier = {
				add = 5
				is_historical_focus_on = no
				has_global_flag = agadir_france_rejected_inital_german_demands
			}
			modifier = {
				add = 5
				is_historical_focus_on = no
				has_global_flag = britain_abandoned_france_in_agadir
			}
			# if negotiations completely fail...
			modifier = {
				add = 100
				has_global_flag = agadir_negotations_failed
			}
		}
	}
}
