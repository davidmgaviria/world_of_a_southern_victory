###################################################
# -- Forming Balkan League
###################################################

balkan_league_cat = {

	league_territorial_negotiations_timer = {
		priority = 25
		icon = generic_political_discourse

		visible = {
			has_country_flag = balkan_league_member
		}
		activation = {
			has_global_flag = balkan_league_territorial_negotiations
			NOT = { has_global_flag = balkan_league_territorial_negotiations_over }
		}
		available = {
			always = no
		}
		cancel_trigger = {
			custom_trigger_tooltip = {
				tooltip = league_needs_two_major_members_tt
				check_variable = { SER.num_major_league_members < 2 }
			}
		}
		cancel_effect = {
			set_global_flag = balkan_league_territorial_negotiations_over
			every_country = {
				limit = { has_country_flag = balkan_league_member }
				clr_country_flag = balkan_league_member
				subtract_from_variable = { SER.num_major_league_members = 1 }
			}
		}

		selectable_mission = no
		days_mission_timeout = 210

		timeout_effect = {
			country_event = wsv_firstbalkan.10
		}
	}

	# -- all the territorial claims
	league_claim_state_x = {
		icon = infiltrate_state
		priority = 10

		allowed = {
			OR = {
				tag = SER
				tag = BUL
				tag = GRE
				tag = MNT
			}
		}
		visible = {
			has_country_flag = balkan_league_member
			has_global_flag = balkan_league_territorial_negotiations
			NOT = { has_global_flag = balkan_league_territorial_negotiations_over }
		}
		available = {
			NOT = {
				ROOT = { has_country_flag = fb_is_claiming_state  }
				FROM = { has_state_flag = state_claimed_by_balkan_member }
			}
		}

		state_target = yes
		on_map_mode = map_and_decisions_view
		target_trigger  = { 
			FROM = {
				# what states are allowed?
				OR = { region = 24  region = 25  region = 26 }
				OR = {
					is_core_of=ROOT
					is_claimed_by=ROOT
				}
				AND = {
					is_owned_by=TUR
					NOT = { state=341 }
				}
				# Has to be near us
				any_neighbor_state = {
					OR = { 
						is_owned_by = ROOT 
						has_state_flag = state_claimed_by_balkan_member
					}
				}
				# BLOCK if: someone else in the Balkan League has a core
				NOT = {
					any_country_with_core = {
						has_country_flag = balkan_league_member
						NOT = { tag = ROOT }   # meaning: it's a different member
					}
				}
			}
		}

		cost = 15
		days_remove = 20
		fire_only_once = yes

		complete_effect = {
			ROOT = { set_country_flag = fb_is_claiming_state }
			FROM = { 
				set_state_flag = state_claimed_by_balkan_member
				save_event_target_as = league_target_state  
			}
			# send event notice to other country that also claim this state, 
			if = {
				limit = {
					NOT = { FROM = { is_core_of=ROOT } }
					any_country = {
						NOT = { tag=ROOT }
						has_country_flag = balkan_league_member  
						FROM={ is_claimed_by=PREV }
					}
				}
				random_country = {
					limit = {
						NOT = { tag=ROOT }
						has_country_flag = balkan_league_member  
						FROM={ is_claimed_by=PREV }
					}
					country_event = { id = wsv_firstbalkan.5   days = 5   random_hours = 110 }
				}
			} 
			else = {
				ROOT = { country_event = { id = wsv_firstbalkan.100  days = 5 } }  # undisputed claim if no one claims
			}
		}

		remove_effect = {
			ROOT = { clr_country_flag = fb_is_claiming_state }
		}

		ai_will_do = {
			factor = 100
		}
	}


	league_withdraw_from_conference = {
		icon = generic_break_treaty
		priority = 0

		allowed = {
			OR = {
				tag = SER
				tag = BUL
				tag = GRE
				tag = MNT
			}
		}
		visible = {
			has_country_flag = balkan_league_member
			has_global_flag = balkan_league_territorial_negotiations
			NOT = { has_global_flag = balkan_league_territorial_negotiations_over }
		}
		available = {
			always = yes
		}

		cost = 0
		days_remove = 0

		complete_effect = {
			# clear our flag + state claims
			leave_balkan_league_effect = yes
			custom_effect_tooltip = leave_league_and_forfit_claims_tt
			# send to fellow league members + update their timers if needed
			every_country = {
				limit = { NOT= { tag = ROOT }  has_country_flag = balkan_league_member }
				country_event =  wsv_firstbalkan.9
				hidden_effect = {
					set_temp_variable = { days_left_for_talks = days_mission_timeout@THIS.league_territorial_negotiations_timer }
					if = {
						limit = { check_variable = { days_left_for_talks < 30 }	}
						add_days_mission_timeout = { mission = THIS.league_territorial_negotiations_timer  days = 30 }
					}
				}
			}
			# if we're Serbia and we leave, collapse faction
			if = {
				limit = { tag = SER }
				custom_effect_tooltip = league_collapses_tt
				hidden_effect = {
					every_country = {
						limit = { has_country_flag = balkan_league_member }
						leave_balkan_league_effect = yes
					}
				}
			}
		}

		ai_will_do = {
			factor = 0
		}
	}
}

# -- First Balkan War
###############################################################################

firstbalkan_war_cat = {

	firstbalkan_peace_implementation_delay = {
		priority = 25
		icon = generic_operation

		visible = {
			OR = {
				tag = FRA
				tag = RUS
				tag = GER
				tag = ENG
				has_country_flag = balkan_league_member
			}
		}
		activation = {
			has_global_flag = firstbalkan_war_ceasefire
			NOT = { has_global_flag = firstbalkan_war_over }
		}
		available = {
			always = no
		}
		cancel_trigger = {
			SER = { has_war_with = TUR }
		}

		days_mission_timeout = 30

		timeout_effect = {
			if = {
				limit = { NOT = { has_global_flag = firstbalkan_war_over } }
				# news event depends on victory
				if = {
					limit = { has_global_flag = firstbalkan_outcome_balkan_league_victory }
					news_event = wsv_firstbalkan.20
				}
				if = {
					limit = { has_global_flag = firstbalkan_outcome_ottoman_victory }
					news_event = wsv_firstbalkan.21
				}
				if = {
					limit = { has_global_flag = firstbalkan_outcome_stalemate }
					news_event = wsv_firstbalkan.22
				}
			}
			set_global_flag = firstbalkan_war_over
		}
	}
}
