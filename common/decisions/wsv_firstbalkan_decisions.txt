# -- Forming Balkan League
###################################################

balkan_league_cat = {

	league_territorial_negotiations_timer = {
		priority = 25
		icon = generic_political_discourse

		visible = {
			has_country_flag = balkan_league_member
		}
		activation = {
			has_global_flag = balkan_league_territorial_negotiations
			NOT = { has_global_flag = balkan_league_territorial_negotiations_over }
		}
		available = {
			always = no
		}
		cancel_trigger = {
			# if two states are fighting, end negotiations
			any_country = {
				has_country_flag = balkan_league_member
				any_enemy_country = {
					has_country_flag = balkan_league_member
				}
			}
		}
		cancel_effect = {
			set_global_flag = balkan_league_territorial_negotiations_over
		}

		selectable_mission = no
		days_mission_timeout = 300

		timeout_effect = {
			every_country = {
				limit = { has_country_flag = balkan_league_member }
				country_event = wsv_firstbalkan.9
			}
		}
	}

	# -- all the territorial claims
	league_claim_state_x = {
		icon = infiltrate_state
		priority = 10

		allowed = {
			OR = {
				tag = SER
				tag = BUL
				tag = GRE
				tag = MNT
			}
		}
		visible = {
			always = yes
			has_country_flag = balkan_league_member
			has_global_flag = balkan_league_territorial_negotiations
			NOT = { has_global_flag = balkan_league_territorial_negotiations_over }
		}
		available = {
			always=yes
			custom_trigger_tooltip = {
				tooltip = league_max_claims_at_a_time_tt
				check_variable = { ROOT.claiming_balkan_states < 2 }
			}
		}

		state_target = yes
		on_map_mode = map_and_decisions_view
		target_trigger  = { 
			FROM = {
				OR = { region = 24  region = 25  region = 26 }
				OR = {
					is_core_of=ROOT
					is_claimed_by=ROOT
				}
				is_owned_by=TUR
				NOT = { state=341 }
			}
		}
		## TODO - hide certain states

		cost = 5
		days_remove = 30

		complete_effect = {
			add_to_variable = { ROOT.claiming_balkan_states = 1 }
			FROM = { save_event_target_as = league_target_state  }
			# send event notice to other countries that also claim this state
			hidden_effect = {
				every_country = {
					limit = { 
						NOT = { tag=ROOT }
						#has_country_flag = balkan_league_member  
						FROM={ is_claimed_by=PREV }
					}
					country_event = { id = wsv_firstbalkan.5   days = 3 }
				}
			}
		}
		remove_effect = {
			subtract_from_variable = { ROOT.claiming_balkan_states = 1 }
		}

		ai_will_do = {
			factor = 100
		}
	}
}
