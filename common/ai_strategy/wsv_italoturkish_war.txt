italy_prepare_libyan_naval_invasion = {
	allowed = {
		original_tag = ITA
	}
	enable = {
		focus_progress = {
			focus = ITA_invade_libya_focus
			progress > 0.01
		}
		NOT = { has_global_flag = italoturkish_war_over }
	}
	abort = {
		OR = {
			has_global_flag = italoturkish_war_over
			AND = {
				controls_state = 450
				controls_state = 448
			}
		}
	}

	ai_strategy = {
		type = prepare_for_war
		id = TUR
		value = 50
	}

	ai_strategy = {
		type = front_control
		state = 450  # Bhengasi
		state = 448  # Tripoli
		ratio = 0.1
		ordertype = front
		execution_type = balanced
		manual_attack = yes
	}

	ai_strategy = {
		type = naval_invasion_focus
		value = 100
	}

}

italy_slowdown_italoturkish_war = {
	allowed = {
		original_tag = TUR
		has_start_date < 1937.1.1
	}
	enable = {
		has_global_flag = italoturkish_war
		has_war_with = TUR
		casualties > 4000
		NOT = { has_global_flag = italoturkish_war_over }
	}
	abort_when_not_enabled = yes

	# be cautious
	ai_strategy = {
		type = front_control
		state_trigger = {
			OR = {
				is_controlled_by = TUR
				is_controlled_by = LBA
			}
			is_on_continent = africa
		}
		priority = 100
		ratio = 0.9  
		ordertype = front
		execution_type = careful
		execute_order = yes
		manual_attack = yes
	}	
}
