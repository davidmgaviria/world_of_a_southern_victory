## League states should press hard against Turks in states they claim -- as long as casulaties arent rough
league_opening_aggressive_moves = {
	enable = {
		has_country_flag = balkan_league_member
		NOT = { has_global_flag = firstbalkan_war_over }
		has_start_date < 1937.1.1
	}
	abort = {
		has_war_with = TUR
		casualties > 20000
	}

	ai_strategy = {
		type = front_control
		state_trigger = {
			is_controlled_by = TUR
			OR = {
				is_claimed_by = ROOT
				is_core_of = ROOT
			}
			NOT = { state = 341 }  # dont rush constantinople, its a bottle neck
		}
		priority = 100
		ratio = 0.4
		ordertype = front
		execution_type = rush_weak
		manual_attack = yes
	}
}
