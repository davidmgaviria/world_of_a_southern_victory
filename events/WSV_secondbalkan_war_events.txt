## Bulgaria unsatisfied with London Treaty
country_event = {
	id = wsv_secondbalkan.1
	title = wsv_secondbalkan.1.t
	desc = wsv_secondbalkan.1.d

	fire_only_once = yes

	trigger = {
		tag = BUL
		has_global_flag = firstbalkan_outcome_balkan_league_victory
		has_global_flag = firstbalkan_war_over
		759 = {
			is_claimed_by = BUL
			is_owned_by = SER
		}
	}
	mean_time_to_happen = { days = 2 }

	# Ask Serbia to return territory
	option = {
		name = wsv_secondbalkan.1.a
		SER = { country_event = { id =  wsv_secondbalkan.2  days = 1 } }
	}
}


## Bulgaria demands Vardar Macedonia
country_event = {
	id = wsv_secondbalkan.2
	title = wsv_secondbalkan.2.t
	desc = wsv_secondbalkan.2.d

	is_triggered_only = yes
	timeout_days = 5
	
	# Fine
	option = {
		name = wsv_secondbalkan.2.a
		trigger = { owns_state = 44 }  # if doesnt have Albania, wont do it

		BUL = { country_event = { id =  wsv_secondbalkan.3  days = 1 } }
		set_country_flag = SER_agreed_to_give_macedonia_to_bul
		ai_chance = { factor = 100 }
	}

	# No way
	option = {
		name = wsv_secondbalkan.2.b
		BUL = { country_event = { id =  wsv_secondbalkan.4  days = 1 } }
		custom_effect_tooltip = likely_lead_to_war_with_bulgaria_tt
	}
}

## Serbia deliberates forming alliance with Greece
country_event = {
	id = wsv_secondbalkan.6
	title = wsv_secondbalkan.6.t
	desc = wsv_secondbalkan.6.d

	fire_only_once = yes

	trigger = {
		tag = SER
		has_global_flag = firstbalkan_war_over
		has_global_flag = BUL_prepares_against_serbia_over_macedonia
		is_in_faction = no
		GRE = {
			has_war = no
			is_in_faction = no
		}
	}
	mean_time_to_happen = { days = 7 }

	# Ask Greece for alliance
	option = {
		name = wsv_secondbalkan.6.a
		GRE = { country_event = { id = wsv_secondbalkan.7  days = 3 } }
	}
}


## Greece asked by Serbia if it wants to form an alliance
country_event = {
	id = wsv_secondbalkan.7
	title = wsv_secondbalkan.7.t
	desc = wsv_secondbalkan.7.d

	is_triggered_only = yes
	timeout_days = 5
	
	# Yes
	option = {
		name = wsv_secondbalkan.7.a
		trigger = { SER = { has_war = no } }

		SER = { country_event = wsv_secondbalkan.8 }
		effect_tooltip = {
			SER = { give_guarantee = GRE }
			GRE = { give_guarantee = SER }
		}
		ai_chance = { 
			factor = 80 
		}
	}

	# Not now
	option = {
		name = wsv_secondbalkan.7.b
		SER = { country_event = wsv_secondbalkan.9 }
		ai_chance = { 
			factor = 20 
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
		}
	}
}

# Second Balkan War Starts
news_event = {
	id = wsv_secondbalkan.20
	title = wsv_secondbalkan.20.t
	desc = wsv_secondbalkan.20.d
	picture = EVENT_hoi4tgw_second_balkan_war

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = secondbalkan_war
		hidden_effect = {
			BUL = {
				declare_war_on = {
					target = SER
					type = take_claimed_state
				}
				add_named_threat = {
					threat = 5
					name = "Second Balkan War"
				}
			}
			GRE = {
				if = {
					limit = { 
						OR = { has_global_flag = secondbalkan_serbian_greek_defensive_alliance   has_guaranteed = SER }
					}
					add_to_war = {
						targeted_alliance = SER
						enemy = BUL
					}
					SER = {
						create_faction = "Anti-Bulgarian Coalition"
						add_to_faction = GRE
					}
				}
			}
			ROM = { country_event = { id = wsv_secondbalkan.21  days = 7 } }
			TUR = { country_event = { id = wsv_secondbalkan.22  days = 10 } }  # joined a little after
		}
	}
	
	option = {
		name = wsv_secondbalkan.20.a
	}
}

		}

			set_country_flag = secondbalkanwar_romanian_intervention
