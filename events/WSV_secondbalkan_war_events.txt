# Second Balkan War Starts
news_event = {
	id = wsv_secondbalkan.15
	title = wsv_secondbalkan.15.t
	desc = wsv_secondbalkan.15.d
	picture = EVENT_hoi4tgw_second_balkan_war

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = secondbalkan_war
		hidden_effect = {
			BUL = {
				declare_war_on = {
					target = SER
					type = take_claimed_state
				}
				add_named_threat = {
					threat = 5
					name = "Second Balkan War"
				}
			}
			GRE = {
				if = {
					limit = { 
						OR = { has_global_flag = secondbalkan_serbian_greek_defensive_alliance   has_guaranteed = SER }
					}
					add_to_war = {
						targeted_alliance = SER
						enemy = BUL
					}
					set_country_flag = antibulgarian_league_member
					SER = {
						create_faction = "Anti-Bulgarian Coalition"
						add_to_faction = GRE
						set_country_flag = antibulgarian_league_member
					}
				}
			}
			ROM = { country_event = { id = wsv_secondbalkan.16  days = 7 } }
			TUR = { country_event = { id = wsv_secondbalkan.17  days = 10 } }  # joined a little after
		}
	}
	
	option = {
		name = wsv_secondbalkan.15.a
	}
}

	}

# Bulgaria - tide of war turned, sue for peace early (OTL)?
country_event = {
	id = wsv_secondbalkan.20
	title = wsv_secondbalkan.20.t
	desc = wsv_secondbalkan.20.d

	fire_only_once = yes

	trigger = {
		tag = BUL
		has_global_flag = secondbalkan_war
		NOT = { has_global_flag = secondbalkan_war_over }
		NOT = { has_country_flag = BUL_requests_peace_secondbalkan }
		# serbia conditions (only fire this if Serbia is leading the 'anti-bulgarian league' or fighting alone
		SER = {
			has_war_with = BUL
			OR = {
				is_in_faction = no
				is_faction_leader = yes
			}
		}
		# lose conditiions
		NOT = { controls_state = 759 }  # vardar macedonia
		NOT = { controls_state = 731 }  # central bulgaria
		NOT = { controls_state = 855 }  # thessalonkia
		surrender_progress > 0
		surrender_progress < 0.4   # if greater than 40%, use the other event
	}

	immediate = {
		set_country_flag = BUL_requests_peace_secondbalkan
	}

	# End the war
	option = { 
		name = wsv_secondbalkan.20.a
		SER = { country_event = { id = wsv_secondbalkan.22  days = 1 } }
		ai_chance = { factor = 100 }
	}

	# Continue the fight
	option = { 
		name = wsv_secondbalkan.20.b
		ai_chance = { factor = 0 }
	}
}

# Anti-Bulgarian League has lost, sue for peace
country_event = {
	id = wsv_secondbalkan.24
	title = wsv_secondbalkan.24.t
	desc = wsv_secondbalkan.24.d

	fire_only_once = yes

	trigger = {
		has_global_flag = secondbalkan_war
		NOT = { has_global_flag = secondbalkan_war_over }
		NOT = { has_global_flag = antibulgarian_league_requests_peace_secondbalkan }
		# who gets the event?
		OR = {
			# serbia is alone
			AND = {
				tag = SER
				has_war_with = BUL
				is_in_faction = no
				surrender_progress > 0.3  # we're losing, give up
			}
			# a member in anti-bulgarian league
			AND = {
				SER = { is_faction_leader = yes }
				has_country_flag = antibulgarian_league_member
				has_war_with = BUL
				surrender_progress > 0.2  # we're getting pushed back
				surrender_progress < 0.6
				# all our allies have fallen
				NOT = {
					any_allied_country = { surrender_progress < 0.6 }
				}
				
			}
		}
	}

	immediate = {
		set_global_flag = antibulgarian_league_requests_peace_secondbalkan
	}

	# End the war
	option = { 
		name = wsv_secondbalkan.24.a
		BUL = { country_event = { id = wsv_secondbalkan.25  days = 1 } }
	}
}

	desc = wsv_secondbalkan.26.d
	picture = EVENT_hoi4tgw_second_balkan_war_end

	major = yes
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			set_global_flag = secondbalkanwar_over
			set_global_flag = secondbalkanwar_bulgarian_victory

			BUL = {
				transfer_state = 759  # we get all of Macedonia
				transfer_state = 106  
				transfer_state = 818
				if = {
					limit = { has_war_with = GRE }
					transfer_state = 731
				}
				if = {
					limit = { has_war_with = ROM }
					transfer_state = 842
				}
			}

			# league peaces out
			SER = {
				white_peace = BUL
				if = { 
					limit = { is_faction_leader = yes }
					dismantle_faction = yes
				}
			}
			every_country = {
				limit = { has_country_flag = antibulgarian_league_member }
				clr_country_flag = antibulgarian_league_member
			}
		}
	}
	
	option = {
		name = wsv_secondbalkan.26.a
	}
}
