# Formation of the Black Hand
country_event = {
	id = wsv_serbia.1
	title = wsv_serbia.1.t
	desc = wsv_serbia.1.d
	
	trigger = {
		tag = SER
		date > 1911.5.9
		NOT = { has_country_flag = SER_black_hand_society_formed }
	}

	mean_time_to_happen = { days = 0 }

	immediate = {
		set_country_flag = SER_black_hand_society_formed
	}

	option = {
		name = wsv_serbia.1.a
		add_ideas = SER_black_hand_org
		complete_national_focus = SER_formation_of_black_hand
	}
}


# Serbian government uncovers Black Hand goals
country_event = {
	id = wsv_serbia.2
	title = wsv_serbia.2.t
	desc = wsv_serbia.2.d
	
	fire_only_once = yes

	trigger = {
		tag = SER
		has_country_flag = SER_black_hand_society_formed
		NOT = { has_country_flag = SER_black_hand_goals_stated }
		is_puppet = no
		has_capitulated = no
	}

	mean_time_to_happen = { days = 35 }

	immediate = {
		set_country_flag = SER_black_hand_goals_stated
		set_country_flag = {
			flag = blackhand_second_goals_delay
			value = 1
			days = 600  # 20 months, so 1.6 years
		}
	}

	option = {
		name = wsv_serbia.2.a
		activate_decision = blackhand_free_northern_slavs
		activate_decision = blackhand_free_southern_slavs
		activate_decision = blackhand_secure_adriatic_port
	}
}
