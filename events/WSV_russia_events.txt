## ------ Zmestvo Bill and effects 

# Zmestvo land bill 'impasse' (Stolypin comes to Tsar to push through the bill)
country_event = {
	id = wsv_russia.1
	title = wsv_russia.1.t
	desc = wsv_russia.1.d

	fire_only_once = yes
	
	trigger = {
		tag = RUS
		has_government = neutrality		# needs to be tsarist
		date > 1911.3.1					# occured in march of 1911
		NOT = { has_country_flag = zmestov_land_bill_crisis_occured }
	}

	mean_time_to_happen = {
		days = 14
	}

	# Allow Stolypin to pass the bill by decree
	option = { 
		name = wsv_russia.1.a
		add_stability = -0.05
		add_popularity = {
			ideology = neutrality
			popularity = 0.05
		}
		set_country_flag = zmestov_land_bill_passed_by_decree
	}

	# Accept Stolypin's resignation
	option = { 
		name = wsv_russia.1.b
		add_political_power = -75
		retire_character = RUS_pyotr_stolypin
		set_country_flag = stolypin_no_longer_advisor
	}
}

## Food shortages hit during war
country_event = {
	id = wsv_russia.105
	title = wsv_russia.105.t
	desc = wsv_russia.105.d
	picture = report_event_generic_breadlines

	trigger = {
		tag = RUS
		has_war_with_major = yes
		# bascially either at war for 1.5 years or 1 year but w/ war economy
		OR = {
			longest_war_length > 17
			AND = {
				longest_war_length > 11		# a year into the war, calculated at Aug 1, 1915
				OR = {
					has_idea = war_economy
					has_idea = tot_economic_mobilisation 
					has_idea = RUS_prioritize_army_supplies
				}
			}
		}
		NOT = { has_idea = generic_wartime_food_shortage_1 }
		NOT = { has_idea = generic_wartime_food_shortage_2 }
		NOT = { has_idea = generic_wartime_food_shortage_3 }
	}

	mean_time_to_happen = { days = 14 }  #cant be days=1 or will cause error

	option = {
		name = wsv_russia.105.a
		add_ideas = generic_wartime_food_shortage_1
		set_country_flag = {
			flag = food_crisis_escalation_countdown
			days = 480  # a little less than 1.5 years after, calculated to Dec 1, 1916
			value = 1
		}
		add_popularity = {
			ideology = communism
			popularity = 0.07
		}
	}
}
