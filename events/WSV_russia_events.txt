## Our patron Serbia was attacked by AUH
country_event = {
	id = wsv_russia.21
	title = wsv_russia.21.t
	desc = wsv_russia.21.d

	trigger = {
		tag = RUS
		has_country_flag = patron_of_serbia
		SER = {
			has_defensive_war_with = AUH
		}
		# bypass if at war with AUH already or in faction/guaranteeing Serbia
		NOT = {
			 has_war_with = AUH
			 is_in_faction_with = SER 
			 has_guaranteed = SER
		}
	}

	mean_time_to_happen = { days = 2 }  #cant be days=1 or will cause error

	## Help Serbia
	option = { 
		name = wsv_russia.21.a
		AUH = { country_event = wsv_russia.22 }
		ai_chance = { factor = 100 }
	}

	## Abandon Serbia
	option = { 
		name = wsv_russia.21.b
		SER = { add_opinion_modifier = { target = ROOT   modifier = faction_traitor } }
		clr_country_flag = patron_of_serbia
		ai_chance = { factor = 0 }
	}
}

## Food shortages hit during war
country_event = {
	id = wsv_russia.55
	title = wsv_russia.55.t
	desc = wsv_russia.55.d
	picture = report_event_generic_breadlines

	trigger = {
		tag = RUS
		has_war_with_major = yes
		# bascially either at war for 1.5 years or 1 year but w/ war economy
		OR = {
			longest_war_length > 17
			AND = {
				longest_war_length > 11		# a year into the war, calculated at Aug 1, 1915
				OR = {
					has_idea = war_economy
					has_idea = tot_economic_mobilisation 
					has_idea = RUS_prioritize_army_supplies
				}
			}
		}
		NOT = { has_idea = generic_wartime_food_shortage_1 }
		NOT = { has_idea = generic_wartime_food_shortage_2 }
		NOT = { has_idea = generic_wartime_food_shortage_3 }
	}

	mean_time_to_happen = { days = 14 }  #cant be days=1 or will cause error

	option = {
		name = wsv_russia.55.a
		add_ideas = generic_wartime_food_shortage_1
		set_country_flag = {
			flag = food_crisis_escalation_countdown
			days = 480  # a little less than 1.5 years after, calculated to Dec 1, 1916
			value = 1
		}
		add_popularity = {
			ideology = communism
			popularity = 0.07
		}
	}
}
