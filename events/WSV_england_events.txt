## Call for volunteers (Kitchner's Army)
country_event = {
	id = wsv_england.100
	title = wsv_england.100.t
	desc = wsv_england.100.d
	#picture = news_event_wsv_parliament

	trigger = {
		tag = ENG
		has_war_with_major = yes
		has_idea = ENG_volunteer_military_ideals
		NOT = { has_global_flag = kitchners_army_is_formed }
	}

	mean_time_to_happen = { days = 3 }

	immediate = {
		set_global_flag = kitchners_army_is_formed
		set_country_flag = {
			flag = ENG_initial_enlistment_wave
			days = 180
			value = 1
		}
	}

	fire_only_once = yes

	option = {
		name = wsv_england.100.a
		add_war_support = 0.1
		add_ideas = ENG_kitcheners_army_idea
	}
}


## Volunteer enthusiaism wanes (6 months in)
country_event = {
	id = wsv_england.101
	title = wsv_england.101.t
	desc = wsv_england.101.d
	#picture = news_event_wsv_parliament

	trigger = {
		tag = ENG
		has_war_with_major = yes
		has_idea = ENG_kitcheners_army_idea
		NOT = { has_country_flag = ENG_initial_enlistment_wave }
	}

	mean_time_to_happen = { days = 3 }

	fire_only_once = yes

	option = {
		name = wsv_england.101.a
		swap_ideas = {
			remove_idea = ENG_kitcheners_army_idea
			add_idea = ENG_kitcheners_army_reduced_idea
		}
	}
}

## News event when Britain introduces conscription
news_event = {
	id = wsv_england.104
	title = wsv_england.104.t
	desc = wsv_england.104.d
	#picture = news_event_wsv_parliament

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = england_introduces_conscription
	}

	# England
	option = {
		name = wsv_england.104.a
		trigger = { tag = ENG }
	}
	 
	# England's allies
	option = {
		name = wsv_england.104.b
		trigger = { is_in_faction_with = ENG }
	}	

	# Englands enemies
	option = {
		name = wsv_england.104.c
		trigger = { has_war_with = ENG }
	}

	# neutrals
	option = {
		name = wsv_england.104.e
		trigger = {
			NOT = { 
				tag = ENG
				has_war_with = ENG 
				is_in_faction_with = ENG 
			}
		}
	}
}
