## War scare over Serbia occupying Albania in 1912 
add_namespace = wsv_1912_warscare


## ------------------------ Prelude to Crisis ------------------------ ##


## Serbia occupies Albania (border fixing event)
country_event = {
	id = wsv_1912_warscare.1
	title = wsv_1912_warscare.1.t
	desc = wsv_1912_warscare.1.d

	trigger = {
		tag = SER
		has_global_flag = firstbalkan_war
		NOT = { has_global_flag = firstbalkan_war_over }
		has_war_with = TUR
		any_country = {
			NOT = { tag = SER }
			is_in_faction_with = SER 
			has_full_control_of_state = 44
		}
	}

	mean_time_to_happen = { days=0 }
	
	option = {
		name = wsv_1912_warscare.50.a
		44 = { set_state_controller_to = SER }
		if = {
			limit = { 
				MNT = { is_in_faction_with = SER } 
				any_country = {
					is_in_faction_with = SER 
					has_full_control_of_state = 866
				}
			}
			866 = { set_state_controller_to = MNT }
		}
	}
}


## Austria gets notif of Serbian encroachment in Albania
country_event = {
	id = wsv_1912_warscare.2
	title = wsv_1912_warscare.2.t
	desc = wsv_1912_warscare.2.d

	fire_only_once = yes   # only fires event once

	trigger = {
		tag = AUH
		has_war = no
		has_global_flag = firstbalkan_war
		SER = { 
			controls_state = 44 
			NOT = { owns_state = 44 }
			NOT = { is_guaranteed_by = AUH }
			NOT = { is_in_faction_with = AUH }
		}
	}

	mean_time_to_happen = { days = 3 }

	# Mobilize against Serbia 
	option = {
		name = wsv_1912_warscare.2.a
		news_event = { id = wsv_1912_warscare.3  days = 2 }
		ai_chance = { factor = 100 }
	}

	# Ignore it
	option = {
		name = wsv_1912_warscare.2.b
		add_political_power = -100
		ai_chance = { factor = 0 }
	}
}


## News event of Austrian mobilization
news_event = {
	id = wsv_1912_warscare.3
	title = wsv_1912_warscare.3.t
	desc = wsv_1912_warscare.3.d
	#picture = EVENT_hoi4tgw_first_balkan_war_end  ## TODO - needs picture

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = 1912_warscare_AUH_mobilized
		hidden_effect = {
			if = {
				limit = { tag = AUH }
				add_named_threat = {
					threat = 5
					name = "Austria-Hungary Mobilizes Over the Albanian Question"
				}
				## TODO - needs to add idea about mobilization cost
			}
		}
	}
	
	# Austrian response
	option = {
		name = wsv_1912_warscare.3.a
		trigger = { tag = AUH }
	}

	# Serbian response
	option = {
		name = wsv_1912_warscare.3.b
		trigger = { tag = SER }
	}

	# Russian response (counter mobilization )
	option = {
		name = wsv_1912_warscare.3.c
		trigger = { tag = RUS }
		news_event = { id = wsv_1912_warscare.4  days = 2 }
	}

	# Rest of the world response
	option = {
		name = wsv_1912_warscare.3.e
		trigger = { 
			NOT = { tag = AUH }
			NOT = { tag = SER }
			NOT = { tag = RUS }
		}
	}
}

### -------------- Initial German Response -------------- ###

## Germany has to decide whether to back AUH or not
country_event = {
	id = wsv_1912_warscare.5
	title = wsv_1912_warscare.5.t
	desc = wsv_1912_warscare.5.d

	fire_only_once = yes

	trigger = {
		tag = GER
		is_in_faction_with = AUH
		is_faction_leader = yes
		has_global_flag = 1912_warscare_AUH_mobilized
		has_global_flag = 1912_warscare_crisis_began
		NOT = { has_global_flag = 1912_warscare_crisis_over }
	}

	mean_time_to_happen = { days=4 }

	# We are not ready for war, urge moderation from AUH
	option = {
		name = wsv_1912_warscare.5.a
		AUH = { country_event = { id = wsv_1912_warscare.6  days = 1 } }
		ai_chance = {
			factor = 80
		}
	}

	# Give them a blank check
	option = {
		name = wsv_1912_warscare.5.b
		trigger = {
			has_war = no
		}
		AUH = { country_event = { id = wsv_1912_warscare.7  days = 1 } }
		custom_effect_tooltip = may_lead_to_major_war_tt
		ai_chance = {
			factor = 15
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
		}
	}
}

### -------------- Initial British / French Response -------------- ###

## Britain / France has to decide whether to back Russia or not
country_event = {
	id = wsv_1912_warscare.15
	title = wsv_1912_warscare.15.t
	desc = wsv_1912_warscare.15.d

	trigger = {
		OR = {
			# is France and in faction with Russia
			AND = {
				tag = FRA
				is_in_faction_with = RUS
			}
			# is Britain and indirectly tied to Russia
			AND = {
				tag = ENG
				FRA = { is_in_faction_with = RUS }
				FRA = { is_guaranteed_by = ENG }
			}
		}
		has_global_flag = 1912_warscare_RUS_mobilized
		has_global_flag = 1912_warscare_crisis_began
		NOT = { has_global_flag = 1912_warscare_crisis_over }
		NOT = { has_country_flag = 1912_warscare_entente_nation_sent_response }
	}

	mean_time_to_happen = { days=4 }

	immediate = {
		set_country_flag = 1912_warscare_entente_nation_sent_response
	}

	# (France) We are not ready for war, urge moderation from RUS
	option = {
		name = wsv_1912_warscare.15.a
		trigger = { tag = FRA }
		set_global_flag = 1912_warscare_france_urged_moderation
	}

	# (Britain) We will not support Russia
	option = {
		name = wsv_1912_warscare.15.b
		trigger = { tag = ENG }
		set_global_flag = 1912_warscare_britain_stepped_back
	}

	# (Both) we will back
	option = {
		name = wsv_1912_warscare.15.c
		set_country_flag = 1912_warscare_entente_nation_promised_support  ## COUNTRY FLAG!
		if = {
			limit = { has_war_support < 0.7 }
			add_war_support = -0.05
			add_political_power = -75
			custom_effect_tooltip = populace_wants_no_war_tt
		}
		else_if = {
			limit = { has_war_support < 0.5 }
			add_war_support = -0.1
			add_political_power = -100
			custom_effect_tooltip = populace_wants_no_war_tt
		}
	}
}
