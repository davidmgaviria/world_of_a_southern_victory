	id = wsv_germany.4
	title = wsv_germany.4.t
	desc = wsv_germany.4.d

	major = yes

		set_global_flag = GER_boosts_military_spending_1913

			GER = {	
				add_named_threat = {
					threat = 4
					name = "Army Expansion Program"
				}
				add_timed_idea = {
					idea = GER_army_law_1913
					days = 720
				}
				remove_ideas =  GER_naval_law_1908
				remove_ideas =  GER_naval_law_1912_low
			}
		}
	}

	option = {
		name = wsv_germany.4.a
		trigger = {
			OR = { tag = GER   is_in_faction_with = GER }
		}
	}

	option = {
		name = wsv_germany.4.b
		trigger = {
			OR = { tag = FRA   tag = RUS }
		}
	}

	option = {
		name = wsv_germany.4.c
		trigger = {
			NOT = { tag = GER   is_in_faction_with = GER }
			NOT = { tag = FRA   tag = RUS }
		}
	}
}

## ------------------------ Germany and Turkey ------------------------ ##

# German overtures to Turkey
country_event = {
	id = wsv_germany.80
	title = wsv_germany.80.t
	desc = wsv_germany.80.d

	is_triggered_only = yes

	# Lets work closer together
	option = {
		name = wsv_germany.80.a
		GER = { 
			country_event = { id = wsv_germany.81  days = 2 } 
			add_opinion_modifier = {
				target = TUR
				modifier = HOI4TGW_Opinion_improved_diplomatic_relations
			}
		}
		# on turkey
		set_country_flag = TUR_moved_closer_to_germany	 	# flag set on turkey
		add_opinion_modifier = {
			target = GER
			modifier = HOI4TGW_Opinion_improved_diplomatic_relations
		}
		ai_chance = { factor = 100 }
	}
	
	# Go away
	option = {
		name = wsv_germany.80.b
		GER = { country_event = { id = wsv_germany.82  days = 2 } }
		ai_chance = { factor = 0 }
	}
}
