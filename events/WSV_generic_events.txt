# ---- Request to join faction -- "Join War" National focus
# Can we join?   ROOT is country recieving request, FROM is sender
country_event = {
	id = wsv_gw_generic.2
	title = wsv_gw_generic.2.t
	desc = wsv_gw_generic.2.d

    is_triggered_only = yes

	# Yes
	option = {
		name = wsv_gw_generic.2.a
		if = {
			limit = { is_faction_leader = yes }
			add_to_faction = FROM
		} else = {
			FACTION_LEADER = { add_to_faction = FROM }
		}
		FROM = { country_event = { id = wsv_gw_generic.3  days = 1 } }
		ai_chance = { factor = 100 }
	}

	# No
	option = {
		name = wsv_gw_generic.2.b
		FROM = { country_event = { id = wsv_gw_generic.4  days = 1 } }
		ai_chance = { factor = 0 }
	}
}

#christmas truce
news_event = {
	id = wsv_gw_generic.103
	title = wsv_gw_generic.103.t
	desc = wsv_gw_generic.103.d
	picture = news_event_wsv_christmas_truce

	mean_time_to_happen = { days = 2 }  #cant be days=1 or will cause error

	major = yes

    trigger = {
    	OR= {
    		GER = { has_war_with = FRA }
    		USA = { has_war_with = CSA }
    	}
    	OR = {
    		AND = { date > 1914.12.23	date < 1915.1.10 }		
    		AND = { date > 1915.12.23	date < 1916.1.10 }		#if war starts in 1915 we need this to trigger it
    	}
		NOT = {
			has_global_flag = christmas_truce 
			GER = { surrender_progress > 0.25 }
			FRA = { surrender_progress > 0.25 }
			ENG = { surrender_progress > 0.25 }
			USA = { surrender_progress > 0.25 }
			CSA = { surrender_progress > 0.25 }
		}
    }

    immediate = { hidden_effect = { set_global_flag = christmas_truce } }
	
	option = {
		name = wsv_gw_generic.103.a
		trigger = { has_war = yes }
	}

	option = {
		name = wsv_gw_generic.103.b
		trigger = { has_war = no }
	}
}
