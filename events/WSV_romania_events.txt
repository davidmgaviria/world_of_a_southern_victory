add_namespace = wsv_romania

# Carol I dies (October 10 1914)
country_event = {
	id = wsv_romania.1
	title = wsv_romania.1.t
	desc = wsv_romania.1.d

	fire_only_once = yes

    trigger = {
		tag = ROM
		has_country_leader = { character = ROM_carol_i  ruling_only = yes }
		date > 1914.10.10
		NOT = { has_global_flag = king_carol_i_of_romania_died }
	}
	
	mean_time_to_happen = { days = 0 }

	immediate = {
		set_global_flag = king_carol_i_of_romania_died
		hidden_effect = {
			kill_country_leader = yes
			promote_character = ROM_ferdinand_i
		}
	}

	option = {
		name = wsv_romania.1.a
		effect_tooltip = {
			promote_character = ROM_ferdinand_i
		}
	}
}
