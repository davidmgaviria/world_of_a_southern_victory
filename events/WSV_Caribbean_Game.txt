## wsv_carib_game.1 -- Simon faces revolt in Haiti (news event for US, CS, Haiti)
news_event = {
	id = wsv_carib_game.1
	title =wsv_carib_game.1.t
	desc = wsv_carib_game.1.d
	#picture =  news_event_south_america_civil_war -- TODO

	trigger = {
		NOT = { has_global_flag = HAI_1911_revolt }
		date > 1911.5.1  
		# does Haiti exist & have Simon?
		HAI = {
			is_puppet = no
			has_war = no
			HAI_francois_antoine_simon = { is_country_leader = yes }
		}
	}
	mean_time_to_happen = { days = 60 }  # some point in july

	immediate = {
		hidden_effect = {
			set_global_flag = HAI_1911_revolt
			HAI = {
				857 = { set_demilitarized_zone = yes } 
				activate_decision = HAI_suppress_1911_revolt_deadline
				country_event = { id =  wsv_haiti.1  days = 5 }   # General Cinncanatus defects
			}
		}
	}

	option = {
		name = wsv_carib_game.1.a
		trigger = { tag = USA }
	}

	option = {
		name = wsv_carib_game.1.b
		trigger = { tag = CSA }
	}

	option = {
		name = wsv_carib_game.1.c
		trigger = { tag = HAI }
		custom_effect_tooltip = HAI_1911_revolt_decisions_tt
	}

	option = {
		name = wsv_carib_game.1.e
		trigger = { 
			NOT = { tag = USA }
			NOT = { tag = CSA }
			NOT = { tag = HAI }
		}
	}
}
