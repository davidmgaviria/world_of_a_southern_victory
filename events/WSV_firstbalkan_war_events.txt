# Country claims x state which we also want
country_event = {
	id = wsv_firstbalkan.5
	title = wsv_firstbalkan.5.t
	desc = wsv_firstbalkan.5.d

	is_triggered_only = yes

	# No we want it too
	option = {
		name = wsv_firstbalkan.5.a
		trigger = {
			event_target:league_target_state = { 
				NOT = { is_core_of = FROM } 
				NOT = { FROM = { tag = MNT } state=866  state=761 }
				NOT = { FROM = { tag = GRE } state=758 }
				NOT = { FROM = { tag = BUL } state=756 state=757  state=758 }
			}
		}
		# set as disputed
		event_target:current_state = {
			set_state_flag = league_disputed_claim  
			if = {
				limit = { OR = { state = 855  state = 731 } }  # thessalonika & central macedonia
				set_variable = { primary_claimer = GRE }
				set_variable = { secondary_claimer = BUL }
			}
			if = {
				limit = { state = 107 }  # northern macedonia
				set_variable = { primary_claimer = SER }
				set_variable = { secondary_claimer = BUL }
			}
			if = {
				limit = { state = 759 }  # east macedonia
				set_variable = { primary_claimer = BUL }
				set_variable = { secondary_claimer = SER }
			}
		}
		FROM = { country_event = { id = wsv_firstbalkan.7  days = 2 } }
		ai_chance = { factor = 100 }
	}

	# Fine let them have it
	option = {
		name = wsv_firstbalkan.5.b
		FROM = { country_event = { id = wsv_firstbalkan.7  days = 2 } }
		# drop our claim
		event_target:league_target_state = {
			remove_claim_by = ROOT
			clr_state_flag = league_disputed_claim  
		}
	}
}

# First Balkan War Starts
news_event = {
	id = wsv_firstbalkan.11
	title = wsv_firstbalkan.11.t
	desc = wsv_firstbalkan.11.d
	picture = EVENT_hoi4tgw_first_balkan_war

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = firstbalkan_war
		hidden_effect = {
			SER = {
				add_named_threat = {
					threat = 5
					name = "First Balkan War"
				}
				declare_war_on = {
					target = TUR
					type = take_claimed_state
				}
			}
			every_country = {
				limit = { has_country_flag=balkan_league_member  is_in_faction_with=SER }
				add_ideas = generic_liberation_of_balkans
				add_to_war = {
					targeted_alliance = SER
					enemy = TUR
				}
			}
		}
	}
	
	option = {
		name = wsv_firstbalkan.11.a
	}
}

				NOT = { controls_state = 761 } # plav
				NOT = { controls_state = 866 } # scutari

# Alt-hist -- Great Powers intervene in First Balkan (Balkan League losing)
country_event = {
	id = wsv_firstbalkan.14
	title = wsv_firstbalkan.14.t
	desc = wsv_firstbalkan.14.d

	trigger = {
		has_global_flag = firstbalkan_war
		NOT = { has_global_flag = firstbalkan_war_ceasefire  }
		NOT = { has_global_flag = firstbalkan_war_over  }
		NOT  = { has_global_flag = firstbalkan_great_powers_intervene }
		OR = {
			tag = RUS
			tag = FRA
			tag = ENG
			tag = GER
		}
		# Balkan League losing (1 member near capitulation, another getting pushed back (or war has gone on long ))
		any_country = {
			OR = { tag=SER  tag=BUL  tag=GRE }
			has_country_flag = balkan_league_member 
			surrender_progress > 0.5
			OR = {
				any_country = {
					AND = {
						OR = { tag=SER  tag=BUL  tag=GRE }
						NOT = { tag = PREV }
						has_country_flag = balkan_league_member 
						surrender_progress > 0.5
					}
				}
				TUR = { war_length_with = { tag = SER   months > 12 } }
			}
		}
	}
	mean_time_to_happen = { days=0 }
	
	option = {
		name = wsv_firstbalkan.14.a
		if = {
			limit = { NOT = { has_global_flag = firstbalkan_great_powers_intervene } }
			set_global_flag = firstbalkan_great_powers_intervene
			set_global_flag = firstbalkan_outcome_ottoman_victory
			hidden_effect = {
				every_country = {
					limit = { OR = { tag=TUR  has_country_flag=balkan_league_memeber } }
					country_event = wsv_firstbalkan.16
				}
				news_event = { id = wsv_firstbalkan.17  days = 3 }
			}
		}
	}
}

# Ceasefire in First Balkan War 
news_event = {
	id = wsv_firstbalkan.17
	title = wsv_firstbalkan.17.t
	desc = wsv_firstbalkan.17.d
	picture = EVENT_hoi4tgw_first_balkan_war_end

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = firstbalkan_war_ceasefire
		hidden_effect = {
			# Balkan Leagues occupy controlled territory if war wasnt a loss
			if = {
				limit = { NOT = { has_global_flag = firstbalkan_outcome_ottoman_victory } }
				every_country = {
					limit = { has_country_flag=balkan_league_member }
					every_controlled_state = {
						limit = {
							OR = { region=24  region=25  region=26 }
							NOT = { state=341 }  # do not occupy Constantinople
							is_owned_by=TUR
						}
						transfer_state_to = PREV
						set_state_flag = firstbalkan_occupied_ottoman_state
					}
				}
			}
			# peace out
			every_country = {
				limit = {
					has_war_with = TUR
					has_country_flag = balkan_league_member
					is_faction_leader = yes
				}
				white_peace = TUR			# peaces out everyone
			}
		}
	}
	
	option = {
		name = wsv_firstbalkan.17.a
	}
}

# End of First Balkan War (Balkan League victory)
news_event = {
	id = wsv_firstbalkan.20
	title = wsv_firstbalkan.20.t
	desc = wsv_firstbalkan.20.d
	picture = EVENT_hoi4tgw_first_balkan_war_end

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = firstbalkan_war_over
		hidden_effect = {
			# transfer back occupied territory to Turkey first
			every_state = {
				limit = { has_state_flag = firstbalkan_occupied_ottoman_state }
				clr_state_flag = firstbalkan_occupied_ottoman_state
				transfer_state_to = TUR
			}

			## transfer cores next to Balkan victors
			every_country = {
				limit = { has_country_flag = balkan_league_member }
				if = { 
					limit = { tag = SER }
					760 = { transfer_state_to = PREV   clr_state_flag = ottoman_state_claimed_by_league }  # kosovo
				}
				if = { 
					limit = { tag = GRE }
					758 = { transfer_state_to = PREV   clr_state_flag = ottoman_state_claimed_by_league }  # epirus
					867 = { transfer_state_to = PREV   clr_state_flag = ottoman_state_claimed_by_league }  # south macedonia
				}
				if = { 
					limit = { tag = BUL }
					788 = { transfer_state_to = PREV   clr_state_flag = ottoman_state_claimed_by_league }  # east rumelia
				}
			}

			## transfer claims next
			every_state = {
				limit = { 
					has_state_flag=ottoman_state_claimed_by_league  
					NOT={ has_state_flag=league_disputed_claim }    
					is_owned_by=TUR 
				}
				save_event_target_as = current_state

				# is the claim contested or not?
				if = {
					limit = { NOT = { has_state_flag = league_disputed_claim } }
					random_country = {
						limit = { has_country_flag = balkan_league_member    event_target:current_state = { is_claimed_by = PREV } }
						transfer_state = event_target:current_state 
					} 
				} else = {
					random_country = {
						limit = { tag = var:current_state.primary_claimer }	
						save_event_target_as = primary_claimer_TAG
					}
					random_country = {
						limit = { tag = var:current_state.secondary_claimer }	
						save_event_target_as = secondary_claimer_TAG
					}
					if = {
						limit = { event_target:primary_claimer_TAG ={ has_country_flag = balkan_league_member } }
						transfer_state_to = event_target:primary_claimer_TAG
					} else_if  = {
						limit = { event_target:secondary_claimer_TAG ={ has_country_flag = balkan_league_member } }
						transfer_state_to = event_target:secondary_claimer_TAG
					}
				}
				clr_state_flag=ottoman_state_claimed_by_league  
				clr_state_flag=league_disputed_claim  
			}
			
			## form albania if agreed upon  # TODO --- form Albania if Serbia agreed to withdraw
			# ALB = {
			# 	transfer_state = 44
			# 	transfer_state = 803
			# 	transfer_state = 866
			# 	AUH = { give_guarantee = ALB }
			# 	ITA = { give_guarantee = ALB }
			# }

			# last step - dismantle faction
			every_country = {
				limit = { has_country_flag = balkan_league_member }
				remove_ideas = generic_liberation_of_balkans
				clr_country_flag = balkan_league_member
			}
			random_country = {
				limit = {
					has_country_flag = balkan_league_member
					is_faction_leader = yes
				}
				dismantle_faction = yes
			}
		}
	}
	
	option = {
		name = wsv_firstbalkan.20.a
	}
}

	title = wsv_firstbalkan.25.t
	desc = wsv_firstbalkan.25.d

	trigger = {
		tag = SER
		has_global_flag = firstbalkan_war
		NOT = { has_global_flag = firstbalkan_war_over }
		has_war_with = TUR
		any_country = {
			is_in_faction_with = SER 
			has_full_control_of_state = 44
		}
	}

	mean_time_to_happen = { days=0 }
	
	option = {
		name = wsv_firstbalkan.25.a
		44 = { set_state_controller_to = SER }
		if = {
			limit = { 
				MNT = { is_in_faction_with = SER } 
				any_country = {
					is_in_faction_with = SER 
					has_full_control_of_state = 866
				}
			}
			866 = { set_state_controller_to = MNT }
		}
	}
}


## Austria gets notif of Serbian encroachment in Albania
country_event = {
	id = wsv_firstbalkan.26
	title = wsv_firstbalkan.26.t
	desc = wsv_firstbalkan.26.d

	fire_only_once = yes

	trigger = {
		tag = AUH
		has_war = no
		has_global_flag = firstbalkan_war
		NOT = { has_global_flag = AUH_opposes_serbian_occupation_of_albania }
		SER = { 
			controls_state = 44 
			NOT = { owns_state = 44 }
			NOT = { is_guaranteed_by = AUH }
			NOT = { is_in_faction_with = AUH }
		}
	}
	mean_time_to_happen = { days = 3 }
	
	# Send warning to serbia
	option = {
		name = wsv_firstbalkan.26.a
		set_global_flag = AUH_opposes_serbian_occupation_of_albania
		SER = {
			country_event = { id = wsv_firstbalkan.27  days = 1 }
		}
		hidden_effect = { 
			# entente notifs
			if = {
				limit = { RUS = { exists=yes  has_war=no  NOT={is_in_faction_with=AUH} } }
				country_event = { id = wsv_firstbalkan.28  days = 3 }
				if = {
					limit = { FRA = { exists=yes  has_war=no  is_in_faction_with=RUS } }
					country_event = { id = wsv_firstbalkan.28  days = 3 }
					if = {
						limit = { ENG = { exists=yes  has_war=no  FRA={is_guaranteed_by=ENG} } }
						country_event = { id = wsv_firstbalkan.28  days = 3 }
					}
				}
				# global news event -- if at least someone responded
				news_event = { id = wsv_firstbalkan.29  days = 7 } 
			}
			# germany notifs
			if = {
				limit = { GER = { exists=yes  has_war=no  is_in_faction_with=AUH } }
				country_event = { id = wsv_firstbalkan.28  days = 3 }
			}
		}
	}
}
