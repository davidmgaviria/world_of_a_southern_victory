# Serbia asks country if they want to partake in Balkan League territorial negotiations

	title = wsv_firstbalkan.1.t
	desc = wsv_firstbalkan.1.d

	is_triggered_only = yes
	
	# Yes (FROM is Serbia, ROOT is country)
	option = {
		name = wsv_firstbalkan.1.a
		set_country_flag = balkan_league_member

			country_event = { id = wsv_firstbalkan.2  days = 1  random_hours = 125 }  # about 5 days randomized

			base = 100

				factor = 0

	id = wsv_firstbalkan.9
	title = wsv_firstbalkan.9.t
	desc = wsv_firstbalkan.9.d

		set_global_flag = balkan_league_territorial_negotiations_over

	option = {
		name = wsv_firstbalkan.9.a
		if = {
			limit = { SER = { is_in_faction = no  has_war = no } }
			create_faction = "Balkan League"
			every_country = {
				limit = { NOT={ tag=SER }  is_in_faction=no   has_country_flag=balkan_league_member  has_war=no }
				SER = { add_to_faction = ROOT }   # root is the current country
			}
		}
		every_country = {
			limit = { has_country_flag=balkan_league_member  is_in_faction=no } 
			clr_country_flag = balkan_league_member
		}
		hidden_effect = {
			TUR = { country_event = wsv_firstbalkan.10 }
			AUH = { country_event = wsv_firstbalkan.10 }
			RUS = { country_event = wsv_firstbalkan.10 }
		}
	}
}

# First Balkan War Starts
news_event = {
	id = wsv_firstbalkan.11
	title = wsv_firstbalkan.11.t
	desc = wsv_firstbalkan.11.d
	picture = EVENT_hoi4tgw_first_balkan_war

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = firstbalkan_war
		hidden_effect = {
			SER = {
				add_named_threat = {
					threat = 5
					name = "First Balkan War"
				}
				declare_war_on = {
					target = TUR
					type = take_claimed_state
				}
			}
			every_country = {
				limit = { has_country_flag=balkan_league_member  is_in_faction_with=SER }
				add_ideas = generic_liberation_of_balkans
				add_to_war = {
					targeted_alliance = SER
					enemy = TUR
					hostility_reason = asked_to_join
				}
			}
		}
	}
	
	option = {
		name = wsv_firstbalkan.11.a
	}
}


# Turkey (notif) informed that war is lost
country_event = {
	id = wsv_firstbalkan.12
	title = wsv_firstbalkan.12.t
	desc = wsv_firstbalkan.12.d

	trigger = {
		has_global_flag = firstbalkan_war
		NOT = { has_global_flag = firstbalkan_war_over  }
		has_war_with = SER
		OR = {
			surrender_progress > 0.25
			AND = {
				NOT = { controls_state = 756 } # eastern thrace
				NOT = { controls_state = 788 } # rumelia
				NOT = { controls_state = 184 } # thrace
				NOT = { controls_state = 855 } # thessalonika

				NOT = { controls_state = 758 } # central epirus
				NOT = { controls_state = 44  } # albania
				NOT = { controls_state = 106 } # macedonia
			}
		}
	}
	fire_only_once = yes
	
	option = {
		name = wsv_firstbalkan.12.a
		SER = { 
			country_event = { id = wsv_firstbalkan.13  days = 3 }
		}
	}
}

# Alt-hist -- Great Powers intervene in First Balkan (Balkan League losing)
country_event = {
	id = wsv_firstbalkan.14
	title = wsv_firstbalkan.14.t
	desc = wsv_firstbalkan.14.d

	trigger = {
		has_global_flag = firstbalkan_war
		NOT = { has_global_flag = firstbalkan_war_over }
		NOT  = { has_global_flag = firstbalkan_great_powers_intervene }
		OR = {
			tag = RUS
			tag = FRA
			tag = ENG
			tag = GER
		}
		# Balkan League losing (1 member near capitulation, another getting pushed back (or war has gone on long ))
		any_country = {
			OR = { tag=SER  tag=BUL  tag=GRE }
			has_country_flag = balkan_league_member 
			surrender_progress > 0.5
			OR = {
				any_country = {
					AND = {
						OR = { tag=SER  tag=BUL  tag=GRE }
						NOT = { tag = PREV }
						has_country_flag = balkan_league_member 
						surrender_progress > 0.5
					}
				}
				TUR = { war_length_with = { tag = SER   months > 12 } }
			}
		}
	}
	
	option = {
		name = wsv_firstbalkan.14.a
		if = {
			limit = { NOT = { has_global_flag = firstbalkan_great_powers_intervene } }
			set_global_flag = firstbalkan_great_powers_intervene
			set_global_flag = firstbalkan_outcome_ottoman_victory
			hidden_effect = {
				every_country = {
					limit = { OR = { tag=TUR  has_country_flag=balkan_league_memeber } }
					country_event = wsv_firstbalkan.16
				}
				news_event = { id = wsv_firstbalkan.17  days = 3 }
			}
		}
	}
}

# Ceasefire in First Balkan War 
news_event = {
	id = wsv_firstbalkan.17
	title = wsv_firstbalkan.17.t
	desc = wsv_firstbalkan.17.d
	picture = EVENT_hoi4tgw_first_balkan_war_end

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = firstbalkan_war_ceasefire
		hidden_effect = {
			# Balkan Leagues occupy controlled territory if war wasnt a loss
			if = {
				limit = { NOT = { has_global_flag = firstbalkan_outcome_ottoman_victory } }
				every_country = {
					limit = { has_country_flag=balkan_league_member }
					every_controlled_state = {
						limit = {
							OR = { region=24  region=25  region=26 }
							NOT = { state=341 }  # do not occupy Constantinople
							is_owned_by=TUR
						}
						transfer_state_to = PREV
						set_state_flag = firstbalkan_occupied_ottoman_state
					}
				}
			}
			# peace out
			every_country = {
				limit = {
					has_war_with = TUR
					has_country_flag = balkan_league_member
					is_faction_leader = yes
				}
				white_peace = TUR			# peaces out everyone
			}
		}
	}
	
	option = {
		name = wsv_firstbalkan.17.a
	}
}

	title = wsv_firstbalkan.26.t
	desc = wsv_firstbalkan.26.d

		tag = AUH
		has_war = no
		has_global_flag = firstbalkan_war
		NOT = { has_global_flag = AUH_opposes_serbian_occupation_of_albania }

			controls_state = 44 
			NOT = { owns_state = 44 }
			NOT = { is_guaranteed_by = AUH }
			NOT = { is_in_faction_with = AUH }
		}
	}

		SER = {
			country_event = { id = wsv_firstbalkan.27  days = 1 }
		}
		hidden_effect = { 
			# entente notifs
			if = {
				limit = { RUS = { exists=yes  has_war=no  NOT={is_in_faction_with=AUH} } }
				country_event = { id = wsv_firstbalkan.28  days = 3 }
				if = {
					limit = { FRA = { exists=yes  has_war=no  is_in_faction_with=RUS } }
					country_event = { id = wsv_firstbalkan.28  days = 3 }
					if = {
						limit = { ENG = { exists=yes  has_war=no  FRA={is_guaranteed_by=ENG} } }
						country_event = { id = wsv_firstbalkan.28  days = 3 }
					}
				}
				# global news event -- if at least someone responded
				news_event = { id = wsv_firstbalkan.29  days = 7 } 
			}
			# germany notifs
			if = {
				limit = { GER = { exists=yes  has_war=no  is_in_faction_with=AUH } }
