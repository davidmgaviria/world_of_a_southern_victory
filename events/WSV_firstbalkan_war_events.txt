# Ceasefire in First Balkan War 
news_event = {
	id = wsv_firstbalkan.18
	title = wsv_firstbalkan.18.t
	desc = wsv_firstbalkan.18.d
	picture = EVENT_hoi4tgw_first_balkan_war_end

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = firstbalkan_war_ceasefire
		hidden_effect = {
			# Balkan Leagues occupy controlled territory if war wasnt a loss
			if = {
				limit = { NOT = { has_global_flag = firstbalkan_outcome_ottoman_victory } }
				every_country = {
					limit = { has_country_flag=balkan_league_member }
					every_controlled_state = {
						limit = {
							OR = { region=24  region=25  region=26 }
							NOT = { state=341 }  # do not occupy Constantinople
							is_owned_by=TUR
						}
						transfer_state_to = PREV
						set_state_flag = firstbalkan_occupied_ottoman_state
					}
				}
			}
			# peace out
			every_country = {
				limit = {
					has_war_with = TUR
					has_country_flag = balkan_league_member
					is_faction_leader = yes
				}
				white_peace = TUR			# peaces out everyone
			}
		}
	}
	
	option = {
		name = wsv_firstbalkan.18.a
	}
}

# End of First Balkan War (Balkan League victory)
news_event = {
	id = wsv_firstbalkan.20
	title = wsv_firstbalkan.20.t
	desc = wsv_firstbalkan.20.d
	picture = EVENT_hoi4tgw_first_balkan_war_end

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = firstbalkan_war_over
		hidden_effect = {
			# transfer back occupied territory to Turkey first
			every_state = {
				limit = { has_state_flag = firstbalkan_occupied_ottoman_state }
				clr_state_flag = firstbalkan_occupied_ottoman_state
				transfer_state_to = TUR
			}

			## transfer states to Balkan victors
			every_country = {
			 	limit = { has_country_flag = balkan_league_member }
				for_each_loop = {
					array = THIS.claimed_balkan_states
					value = s
					break = break
					transfer_state = s
				}
			}
			every_state = {
				limit = { has_state_flag = state_claimed_by_balkan_member }
				clr_state_flag = state_claimed_by_balkan_member
			}

			## form Albania if Serbia agreed to withdraw
			if = {
				limit = { SER = {  has_country_flag = SER_dropped_claims_on_albania } }
				ALB = {
					transfer_state = 44
					transfer_state = 803
					transfer_state = 866
					AUH = { give_guarantee = ALB }
					ITA = { give_guarantee = ALB }
					set_global_flag = birth_of_albania
				}
				# serbia takes Vadar Macedonia in response
				SER = {
					transfer_state = 759
				}
			}

			# last step - dismantle faction
			every_country = {
				limit = { has_country_flag = balkan_league_member }
				remove_ideas = generic_liberation_of_balkans
				leave_balkan_league_effect = yes
			}
			random_country = {
				limit = {
					has_country_flag = balkan_league_member
					is_faction_leader = yes
				}
				dismantle_faction = yes
			}
		}
	}
	
	option = {
		name = wsv_firstbalkan.20.a
	}
}
