# Serbia contacts Bulgaria about forming an alliance against Turkey (FROM is Serbia, ROOT is Bulgaria)
country_event = {
	id = wsv_firstbalkan.1
	title = wsv_firstbalkan.1.t
	desc = wsv_firstbalkan.1.d

	is_triggered_only = yes
	
	# Yes -- with conditions (Bulgaria wants assurances to part of Macedonia)
	option = {
		name = wsv_firstbalkan.1.a
		SER = { country_event = { id = wsv_firstbalkan.3  days = 3 } }
		ai_chance = { 
			base = 100 
			modifier={
				factor = 0 
				has_war = yes
			}
		}
	}

	# No we wont ally with you
	option = {
		name = wsv_firstbalkan.1.b
		if = {
			limit = { has_war = no } # nationalist will be angered if we refuse for no reason
			add_war_support = -0.1   
			add_political_power = -75
		}
		SER = { 
			country_event = { id = wsv_firstbalkan.2  days = 3 } 
			effect_tooltip = {
				add_opinion_modifier = {
					target = BUL
					modifier = wsv_dislikes_country
				}
			}
		}
		ai_chance = { 
			base = 0 
			modifier={
				add = 1 
				has_war = yes
			}
		}
	}
}

# End of First Balkan War (Balkan League victory)
news_event = {
	id = wsv_firstbalkan.31
	title = wsv_firstbalkan.31.t
	desc = wsv_firstbalkan.31.d
	picture = EVENT_hoi4tgw_first_balkan_war_end

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = firstbalkan_war_over
		hidden_effect = {
			# transfer back occupied territory to Turkey first
			every_state = {
				limit = { has_state_flag = firstbalkan_occupied_ottoman_state }
				clr_state_flag = firstbalkan_occupied_ottoman_state
				transfer_state_to = TUR
			}

			## transfer states to Balkan victors
			every_country = {
			 	limit = { has_country_flag = balkan_league_member }
				for_each_loop = {
					array = THIS.claimed_balkan_states
					value = s
					break = break
					transfer_state = s
				}
			}
			every_state = {
				limit = { has_state_flag = state_claimed_by_balkan_member }
				clr_state_flag = state_claimed_by_balkan_member
			}

			## form Albania if Serbia agreed to withdraw
			if = {
				limit = { SER = {  has_country_flag = SER_dropped_claims_on_albania } }
				ALB = {
					transfer_state = 44
					transfer_state = 803
					transfer_state = 866
					AUH = { give_guarantee = ALB }
					ITA = { give_guarantee = ALB }
					set_global_flag = birth_of_albania
				}
				# serbia takes Vadar Macedonia in response
				SER = {
					transfer_state = 759
				}
			}

			# last step - dismantle faction
			every_country = {
				limit = { has_country_flag = balkan_league_member }
				remove_ideas = generic_liberation_of_balkans
				leave_balkan_league_effect = yes
			}
			random_country = {
				limit = {
					has_country_flag = balkan_league_member
					is_faction_leader = yes
				}
				dismantle_faction = yes
			}
		}
	}
	
	option = {
		name = wsv_firstbalkan.31.a
	}
}
