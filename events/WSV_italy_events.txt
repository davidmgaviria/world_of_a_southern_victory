# Italian invasion of Libya - Global News
news_event = {
	id = wsv_italy.2
	title = wsv_italy.2.t
	desc = wsv_italy.2.d
	picture = EVENT_hoi4tgw_italo_turkish_war

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = italoturkish_war
	    hidden_effect = {
			ITA = {
				set_country_flag = italian_libya_war
				add_named_threat = {
					threat = 3
					name = "Italo-Turkish War"
				}
				transfer_state = 869 # tobruk
				transfer_state = 450 # benghasi
				transfer_state = 448 # tripoli
				load_oob=ITA_italoturkish_inital_landing
        		activate_decision = italoturkish_expedition_first_deadline
				add_ideas = ITA_conquest_of_libya
				set_country_flag = {
					flag = ITA_italoturkisk_delay_for_dodecanese
					value = 1 
					days = 210
				}
	        }
			TUR = {
				load_oob=TUR_italoturkish_garrison
			}
			# libyan theatre
			every_state = {
				limit = { OR = { state=448 state=449 state=450 state=451 state=869 state=870  state=661  state=662  state=663 } }
				set_state_flag = libyan_theatre_state
			}
	    }
	}
		
	option = {
		name = wsv_italy.2.a
		trigger = { TAG = ITA }
	}
	option = {
		name = wsv_italy.2.b
		trigger = { TAG = TUR }
	}
	option = {
		name = wsv_italy.2.c
		trigger = { 
			NOT = { TAG = ITA }
			NOT = { TAG = TUR }
		}
	}
}

	desc = wsv_italy.7.d
	
	fire_only_once = yes

		tag = TUR
		has_global_flag = italoturkish_war
		NOT = { has_global_flag = italoturkish_war_over }
		ITA = {
			controls_state = 869 # tobruk
			controls_state = 450 # benghasi
			controls_state = 448 # tripoli
		}
		# either First Balkan or lost all the territory
		OR = {
			AND = {
				has_global_flag = firstbalkan_war

			}
			NOT = {
				any_owned_state = { has_state_flag = libyan_theatre_state }
			}
		}
	}

	option = {
		name = wsv_italy.7.a
		ITA = { country_event = { id = wsv_italy.8   days = 2 } }
	}
}
