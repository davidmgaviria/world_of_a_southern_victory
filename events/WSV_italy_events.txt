	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = italoturkish_war

			ITA = {
				set_country_flag = italian_libya_war
				add_named_threat = {
					threat = 3
					name = "Italo-Turkish War"
				}
				transfer_state = 869 # tobruk
				transfer_state = 450 # benghasi
				transfer_state = 448 # tripoli
				load_oob=ITA_italoturkish_inital_landing
        		activate_decision = italoturkish_expedition_first_deadline
				add_ideas = ITA_conquest_of_libya
				set_country_flag = {
					flag = ITA_italoturkisk_delay_for_dodecanese
					value = 1 
					days = 210
				}
	        }
			TUR = {
				load_oob=TUR_italoturkish_garrison
			}
			# libyan theatre
			every_state = {
				limit = { OR = { state=448 state=449 state=450 state=451 state=869 state=838  state=661  state=662  state=663 } }
				set_state_flag = libyan_theatre_state
			}
	    }
	}
		
	option = {
		name = wsv_italy.25.a
		trigger = { TAG = ITA }
	}
	option = {
		name = wsv_italy.25.b
		trigger = { TAG = TUR }
	}
	option = {
		name = wsv_italy.25.c
		trigger = { 
			NOT = { TAG = ITA }
			NOT = { TAG = TUR }
		}
	}
}
