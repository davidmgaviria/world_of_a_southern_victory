###=====================================================================================###
###---------------------------- Start of Italo-Turkish War -----------------------------###
###=====================================================================================###

# Italian invasion of Libya - Global News
news_event = {
	id = wsv_italoturkish.1
	title = wsv_italoturkish.1.t
	desc = wsv_italoturkish.1.d
	picture = news_event_italoturkish_war_outbreak

	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = italoturkish_war
	    hidden_effect = {
			ITA = {
				set_country_flag = italian_libya_war
				add_named_threat = {
					threat = 4
					name = "Italo-Turkish War"
				}
				declare_war_on = {
					target = TUR
					type = take_claimed_state
				}
				set_state_controller = 450 # benghasi
				set_state_controller = 448 # tripoli
				load_oob=ITA_italoturkish_inital_landing
        		activate_decision = italoturkish_expedition_first_deadline
				# set this idea as without a time
				remove_ideas = ITA_amphibious_war_effort
				add_ideas = ITA_amphibious_war_effort
	        }
			TUR = {
				add_ideas = TUR_cost_of_war_in_libya
				load_oob=TUR_italoturkish_garrison
				create_faction = "Ottoman Empire"
			}
	    }
	}
		
	option = {
		name = wsv_italoturkish.1.a
		trigger = { TAG = ITA }
	}
	option = {
		name = wsv_italoturkish.1.b
		trigger = { TAG = TUR }
	}
	option = {
		name = wsv_italoturkish.1.c
		trigger = { 
			NOT = { TAG = ITA }
			NOT = { TAG = TUR }
		}
	}
}



# Senussi order joins the war (originally triggered on Libya, then sent to Italy and Turkey)
country_event = {
	id = wsv_italoturkish.2
	title = wsv_italoturkish.2.t
	desc = wsv_italoturkish.2.d

	trigger = {
		tag = LBA
		has_global_flag = italoturkish_war
		NOT = { has_global_flag = italoturkish_war_over }
		TUR = { has_war_with = ITA }
		NOT = { has_global_flag = senussi_joined_italoturkish_war }
	}

	# Italy and Turkey also get notice
	major = yes
	show_major = { 
		OR = { tag = ITA  tag = TUR }
	}
	
	immediate = {
		set_global_flag = senussi_joined_italoturkish_war
	}

	mean_time_to_happen = { days = 5 }

	option = {
		name = wsv_italoturkish.2.a
		trigger = { tag = LBA }
		add_war_support = 0.1
		load_oob = "LBA_italoturkish_senusi_mujahadeen"
		add_to_war = {
			targeted_alliance = TUR
			enemy = ITA
		}
	}

	option = {
		name = wsv_italoturkish.2.b
		trigger = { tag = ITA }
	}

	option = {
		name = wsv_italoturkish.2.c
		trigger = { tag = TUR }
	}
}

# Turkish event - Italy takes all of Libya OR First Balkan erupts
country_event = {
	id = wsv_italoturkish.15
	title = wsv_italoturkish.15.t
	desc = wsv_italoturkish.15.d
	
	fire_only_once = yes
	trigger = {
		tag = TUR
		has_global_flag = italoturkish_war
		NOT = { has_global_flag = italoturkish_war_over }

		ITA = {
			controls_state = 869 # tobruk
			controls_state = 450 # benghasi
			controls_state = 448 # tripoli
		}
		# either lost all territory in Libya, or First Balkan War started
		OR = {
			# lost all territory in Libya
			NOT = {
				any_owned_state = {
					region = 225
					is_controlled_by_ROOT_or_ally = yes
				}	
			}
			# First Balkan started
			AND = {
				has_global_flag = firstbalkan_war
				NOT = { has_global_flag = firstbalkan_war_over }
			}
		}
	}

	mean_time_to_happen = { days = 4 }  

	option = {
		name = wsv_italoturkish.16.a
		ITA = { country_event = { id = wsv_italoturkish.16   days = 2 } }
	}
}


# Italy told Turkey wants to surrender

# Italo-Turkish War - Italy Wins (Global News)
news_event = {
	id = wsv_italoturkish.17
	title = wsv_italoturkish.17.t
	desc = wsv_italoturkish.17.d
	picture = GFX_news_event_generic_sign_treaty2
	
	major = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = italoturkish_war_over
		hidden_effect = {
			ITA = {
				set_country_flag = victory_in_libya
				white_peace = TUR
				every_state = {
					limit = { is_owned_by = TUR  OR = { region = 225  state = 273 } }
					transfer_state_to = ITA
				}
				delete_unit_template_and_units = { division_template = "Libya Coloniali Troops" disband = no }
			}
			TUR = {
				set_country_flag = defeat_in_libya
				dismantle_faction = yes
				delete_unit_template_and_units = { division_template = "Düzenli Kuvvetler" disband = no }
				delete_unit_template_and_units = { division_template = "Libya Milisleri" disband = no }
			}
			LBA = {
				transfer_state = 754
				transfer_state = 872
				delete_unit_template_and_units = { division_template = "Senusi Mücahitleri" disband = yes }
			}
		}
	}
		
	option = {
		name = wsv_italoturkish.17.a
		trigger = { 
			NOT = { TAG = ITA }
			NOT = { TAG = TUR }
		}
	}
	option = {
		name = wsv_italoturkish.17.b
		trigger = { 
			TAG = ITA
		}
	}
	option = {
		name = wsv_italoturkish.17.c
		trigger = { 
			TAG = TUR
		}
	}
}

	id = wsv_italoturkish.25

		TUR = { country_event = { id = wsv_italoturkish.26  days = 3 } }
	}
}


# Italy requests Turkey for compromise peace 
country_event = {
	id = wsv_italoturkish.26
	title = wsv_italoturkish.26.t
	desc = wsv_italoturkish.26.d
	
	is_triggered_only = yes

	# Turks agree
	option = {
		name = wsv_italoturkish.26.a
		news_event = { id = wsv_italoturkish.28  days = 3 }
		custom_effect_tooltip = italoturkish_compromise_peace_tt
		ai_chance = { factor = 1 }
	}

	# Turks refuse (can only do if they're winning)
	option = {
		name = wsv_italoturkish.26.b
		trigger = {  # controls 2 of 3 major costal cities / towns
			italoturkish_controls_two_coastal_cities = yes
		}
		ITA = { country_event = { id = wsv_italoturkish.27  days = 3 } }
		ai_chance = { factor = 100 }
	}
}


# Turkey refused peace
country_event = {
	id = wsv_italoturkish.27
	title = wsv_italoturkish.27.t
	desc = wsv_italoturkish.27.d
	
	is_triggered_only = yes

	# if italy still has time, war continues
	option = {
		name = wsv_italoturkish.27.a
		trigger = { NOT = { has_country_flag = italy_no_time_left_for_libya } }
		add_war_support = -0.05
	}

	# if Italy has no time left, they just lose
	option = {
		name = wsv_italoturkish.27.b
		trigger = { has_country_flag = italy_no_time_left_for_libya }
		news_event = { id = wsv_italoturkish.21  days = 3 } 
	}
}
