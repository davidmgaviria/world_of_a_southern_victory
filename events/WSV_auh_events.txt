## Invade Montenegro
country_event = {
	id = wsv_auh.2
	title = wsv_auh.2.t
	desc = wsv_auh.2.d

	trigger = {
		tag = AUH
		NOT = { has_country_flag = invade_montenegro }
		# if serbia hasnt fallen
		SER = {
			war_length_with = {
				tag = AUH
				months > 5
			}
			surrender_progress < 0.6
		}
		MNT = {
			NOT = { has_war_with = AUH }
			is_puppet = no
			is_in_faction = no
		}
	}

	fire_only_once = yes

	immediate = {
		set_country_flag = invade_montenegro
	}
					
	option = {
		name = wsv_auh.2.a
		create_wargoal = {
			type = puppet_wargoal_focus
			target = MNT
		}
	}
}

## Food shortages hit during war
country_event = {
	id = wsv_auh.20
	title = wsv_auh.20.t
	desc = wsv_auh.20.d
	#picture = report_event_wsv_gathering_protest  # TODO - picture

	trigger = {
		tag = AUH
		has_war_with_major = yes
		# bascially either at war for 1.5 years or 1 year but w/ war economy
		OR = {
			longest_war_length > 17
			AND = {
				longest_war_length > 11
				OR = {
					has_idea = war_economy
					has_idea = tot_economic_mobilisation
				}
			}
		}
		NOT = { has_idea = generic_wartime_food_crisis_1 }
		NOT = { has_idea = generic_wartime_food_crisis_2 }
	}

	mean_time_to_happen = { days = 14 }  #cant be days=1 or will cause error

	option = {
		name = wsv_auh.20.a
		add_ideas = generic_wartime_food_crisis_1
		set_country_flag = {
			flag = food_crisis_escalation_countdown
			days = 500  # about 16 months
			value = 1
		}
	}
}

## -- Death of Franz Joseph
news_event = {
    id = wsv_auh.25
    title = wsv_auh.25.t
    desc = wsv_auh.25.d
    picture = ww1_austria_1

    major = yes

    trigger = {
        AUH = {
            NOT = { has_country_flag = emperor_franz_dead }
            date > 1916.11.21
            date < 1937.1.1
        }
    }

	mean_time_to_happen = {
        days = 0
    }

    immediate = {
        hidden_effect = {
            AUH = {
                kill_country_leader = yes
                create_country_leader={ name = "Karl I" desc=AUH_karl_i_DESC picture="gfx/portraits/AUH/country_leaders/AUH_karl_i.dds" expire="1935.1.1" ideology=monarchy  traits={  } }
                set_country_flag = emperor_franz_dead
            }
        }
    }

    option = {
        name = ww1_austria.1.a
        trigger = { TAG = AUH }
    }

    option = {
        name = ww1_austria.1.b
        trigger = {
            NOT = { TAG = AUH }
        }
    }
}

## News event of AUH collapse
news_event = {
	id = wsv_auh.50
	title = wsv_auh.50.t
	desc = wsv_auh.50.d
	picture = news_event_auh_collapses

	major = yes

	trigger = {
		tag = AUH
		has_idea = AUH_house_of_cards	# will only have during wartime
		has_stability < 0.11
		NOT = { has_global_flag = auh_collapsed }
	}

	mean_time_to_happen = {
		days = 2  # needed to prevent error
	}

	immediate = {
		set_global_flag = auh_collapsed
		hidden_effect = {
			## Czechslovakia
			CZE = {
				transfer_state = 9
				transfer_state = 75
				inherit_technology = AUH
			}
			transfer_units_fraction= {		
				target = CZE
				size = 0.13
				stockpile_ratio = 0.13
				army_ratio = 0.13
				navy_ratio = 0
				air_ratio = 0.04
			}
			## South Slavs
			CRO = {
				transfer_state = 103
				transfer_state = 109
				inherit_technology = AUH
			}
			transfer_units_fraction= {
				target = CRO
				size = 0.05
				stockpile_ratio = 0.05
				navy_ratio = 0
			}
			## Hungary
			HUN = {
				transfer_state = 155
				transfer_state = 43
				transfer_state = 154
				# non core states
				transfer_state = 664
				transfer_state = 70
				transfer_state = 71
				transfer_state = 73
				transfer_state = 76
				transfer_state = 83
				transfer_state = 84
				transfer_state = 82
				transfer_state = 764
				transfer_state = 45
				inherit_technology = AUH
			}
			# transfer troops
			transfer_units_fraction= {
				target = HUN
				size = 0.4
				stockpile_ratio = 0.4
				army_ratio = 0.45		# hungarian forces made up 35%, but only 80% of the empire is left now
				navy_ratio = 0
				air_ratio = 0.3
			}
			# sue for peace
			country_event = { id = wsv_auh.51  days = 5 }
		}
	}

	option = {
		name = wsv_auh.50.a
	}
}
