# Hidden seperatism trigger

country_event = {
	id = wsv_auh.1
	title = wsv_auh.1.t
	desc = wsv_auh1.d

	trigger = {
		tag = AUH
		has_war_with_major = no
		NOT = { has_country_flag = seperatist_trigger_event_cooldown }
		check_variable = { AUH.active_seperatist_groups > 0 }
	}

	mean_time_to_happen = { days = 0 }

	hidden = yes   # AI takes care of choice

	immediate = {
		add_to_variable = { AUH.active_seperatist_missions = 1 }
		if = {
			# if we can still pick another group (up to 2 max) do so
			limit = {
				check_variable = { AUH.active_seperatist_missions < 2 }
				check_variable = { AUH.active_seperatist_missions < AUH.active_seperatist_groups }
			}
			country_event = { id = wsv_auh.1  hours = 1 }  # select another group
		} else = {
			# cooldown flag
			set_country_flag = {
				flag = seperatist_trigger_event_cooldown
				value = 1
				days = 120
			}
		}
	}
	
	option = {  # Italian seperatists
		name = wsv_auh.1.a
		trigger = {
			NOT = { has_active_mission = AUH_italian_seperatists }
			OR = { has_full_control_of_state = 736  has_full_control_of_state = 39 }
		}
		activate_mission = AUH_italian_seperatists
		ai_chance = { factor = 20 }
	}
	option = {  # Galician seperatists
		name = wsv_auh.1.b
		trigger = {
			NOT = { has_active_mission = AUH_galician_seperatists }
			OR = { has_full_control_of_state = 88  has_full_control_of_state = 89  has_full_control_of_state = 91 }
		}
		activate_mission = AUH_galician_seperatists
		ai_chance = { factor = 20 }
	}
	option = { # Romanian seperatists
		name = wsv_auh.1.c
		trigger = {
			NOT = { has_active_mission = AUH_romanian_seperatists }
			OR = { has_full_control_of_state = 84  has_full_control_of_state = 82  has_full_control_of_state = 83   has_full_control_of_state = 76 }
		}
		activate_mission = AUH_romanian_seperatists
		ai_chance = { factor = 20 }
	}
	option = { 	# Bosnian rebels
		name = wsv_auh.1.e
		trigger = {
			NOT = { has_active_mission = AUH_bosnian_rebels }
			104 = {
				is_owned_and_controlled_by = ROOT
				NOT = { is_core_of = ROOT }
			}
		}
		activate_mission = AUH_bosnian_rebels
		ai_chance = { factor = 40 }
	}
}

country_event = {
	id = wsv_auh.27
	title = wsv_auh.27.t
	desc = wsv_auh.27.d

	trigger = {
		tag = AUH
		has_war_with_major = yes

		OR = {
			date > 1917.5.20  				 # historical
			longest_war_length > 36 		 # 3 years
			has_stability < 0.35  			 # stability is plumetting
		}
		NOT = { has_country_flag = AUH_summoned_imperial_council_1917 }
		NOT = { has_country_flag = AUH_cooldown_to_summon_imperial_council }
	}


	mean_time_to_happen = { days = 0 }

	immediate = {
		if = {
			limit = { NOT = { has_variable = ROOT.rejected_council_summon } }
			set_variable = { ROOT.rejected_council_summon = 0 }
		}
	}

	# Agree
	option = {
		name = wsv_auh.27.a
		country_event = { id = wsv_auh.28  days = 10 }
		set_country_flag = AUH_summoned_imperial_council_1917
	}

	# Refuse
	option = {
		name = wsv_auh.27.a
		# only Franz joseph can refuse
		trigger = { 
			has_country_leader = { character = AUH_franz_joseph_i  ruling_only = yes }
		}
		set_country_flag = {
			flag = AUH_cooldown_to_summon_imperial_council
			value = 1
			days = 180
		}
		add_to_variable = { ROOT.rejected_council_summon = 1 }
		country_event = { id = wsv_auh.29 days = 2 }
	}
}
