add_namespace = wsv_dr


# 1911 assassination of Ramón Cáceres
country_event = {
	id = wsv_dr.1
	title = wsv_dr.1.t
	desc = wsv_dr.1.d
	#picture =  report_event_wsv_soldiers_running TODO - needs pic

	trigger = {
		NOT = { has_global_flag = DOM_ramon_caceres_assassinated }
		date > 1911.11.1  
		# does DR exist & have Cáceres?
		original_tag = DOM
		is_puppet = no
		has_war = no
		DOM_ramon_caceres = { is_country_leader = yes }
	}

	mean_time_to_happen = { days = 18 }  # Nov 19, 1911 historically

	immediate = {
		hidden_effect = {
			DOM = {
				set_global_flag = DOM_ramon_caceres_assassinated 
				retire_character = DOM_ramon_caceres
				retire_character = DOM_luis_tejera
				promote_character = DOM_alfredo_victoria
				country_event = { id = wsv_dr.2   days = 17  random_hours = 144 }   # civil war
			}
		}
	}

	option = {
		name = wsv_dr.1.a	
	}
}



# Civil war after presidential assassination
country_event = {
	id = wsv_dr.2
	title = wsv_dr.2.t
	desc = wsv_dr.2.d
	#picture =  report_event_wsv_soldiers_running TODO - needs pic

	is_triggered_only = yes

	# side with the government
	option = {
		name = wsv_dr.2.a
		custom_effect_tooltip = DOM_1911_civil_war_tt
		hidden_effect = {
			set_global_flag = DOM_1911_civil_war
			DOM = {
				start_civil_war = {
					ideology = democratic 
					size = 0   # dont get anything, we'll manually transfer stuff
					capital = 858
					states = {
						858
					}
					# Effects for rebels
					set_cosmetic_tag = DOM_rebel_side
					load_oob = "DOM_1911_rebel_troops"
					recruit_character = DOM_desiderio_arias
					recruit_character = DOM_horacio_vasquez
					promote_character = DOM_horacio_vasquez

					set_popularities = {
						democratic = 60
						neutrality = 38
						fascism = 1
						communism = 1
					}
					remove_ideas = latam_cauldillo_ruler

					# Effect for original Dominican government (under Victoria)
					PREV = {
						# set Eladio as president
						recruit_character = DOM_eladio_victoria
						promote_character = DOM_eladio_victoria
						remove_country_leader_role = { character = DOM_alfredo_victoria  ideology = military_dictatorship }
		
						# remove old generals / leaders
						set_cosmetic_tag = DOM_government_side
						load_oob = "DOM_1911_government_troops"
						DOM_horacio_vasquez	= {
							remove_country_leader_role = { character = THIS  ideology = liberalism }
						}
						DOM_desiderio_arias	= { 
							remove_unit_leader_role = yes   
						}
					}
				}
			}
			news_event = wsv_carib_game.10
		}
	}
}


## Dominican civil war ended
country_event = {
	id = wsv_dr.3
	title = wsv_dr.3.t
	desc = wsv_dr.3.d

	trigger = {
		original_tag = DOM
		has_global_flag = DOM_1911_civil_war
		NOT = { has_global_flag = DOM_1911_civil_war_over }
		has_civil_war = no
		is_puppet = no
	}

	mean_time_to_happen = { days = 2 } 

	immediate = {
		hidden_effect = {
			set_global_flag = DOM_1911_civil_war_over
		}
	}

	# Government won
	option = {
		name = wsv_dr.1.a	
		trigger = {
			has_cosmetic_tag = DOM_government_side
		}
		set_country_flag = DOM_government_won_1911_civil_war
		drop_cosmetic_tag = yes
		hidden_effect = {
			news_event = wsv_carib_game.11
		}
	}

	# Rebels won
	option = {
		name = wsv_dr.1.b
		trigger = {
			has_cosmetic_tag = DOM_rebel_side
		}
		set_country_flag = DOM_rebels_won_1911_civil_war
		drop_cosmetic_tag = yes
		hidden_effect = {
			news_event = wsv_carib_game.12
		}
	}
}
