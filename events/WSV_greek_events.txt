### ------ Venizelos invites the Entente to Salonika ------ ###

# Hidden Greek event to send invitation to either France or Britain
country_event = {
	id = wsv_greece.51
	title = wsv_greece.51.t
	desc = wsv_greece.51.d

	fire_only_once = yes
	hidden = yes

    trigger = {
		tag = GRE
		has_war = no
		is_in_faction = no
		is_puppet = no
		# serbia is falling
		SER = {
			has_war_with = AUH
			surrender_progress > 0.65
			OR = {
				is_in_faction_with = FRA
				is_in_faction_with = ENG
			}
		}
	}

	mean_time_to_happen = { days = 0 }

	option = {
		name = wsv_greece.51.a
		SER = { 
			FACTION_LEADER = {  country_event = wsv_greece.52 }   # send to Serbia' faction leader
		}
	}
}


# Target country (France or Britain) get invitation

	id = wsv_greece.52

	}

	# Decline offer
	option = {
		name = wsv_greece.52.b

	}
}

## Greek King overthrown + reunification of Greece
country_event = {
	id = wsv_greece.4
	title = wsv_greece.4.t
	desc = wsv_greece.4.d

    trigger = {
		tag = GRE
		country_exists = GNU
		NOT = { has_war_with = GNU }
		OR = {
			# if in faction together merge
			is_in_faction_with = GNU 	
			# France was unable to overthrow the Greek king due to Russia
			AND = {
				FRA = { 
					is_in_faction_with = GNU
					OR = {
						has_global_flag = tsar_of_russia_abdicated
						NOT = { is_in_faction_with = RUS }
					}
				}
			}
		}
	}

	mean_time_to_happen = { days = 70 }

	# For Greece
	option = {
		name = wsv_greece.4.a
		set_global_flag = greece_reunited
		retire_country_leader = yes
		hidden_effect = { 
			## add greece to GNU faction
			GNU = {
				if = {
					limit = { is_in_faction = yes }
					faction_leader = { add_to_faction = GRE }
				}
			}
			## have greece eat the GNU
			GRE = {
				annex_country = {
					target = GNU
					transfer_troops = yes
				}
			}
		}
		custom_effect_tooltip = greece_forced_into_entente_tt
	}
}
