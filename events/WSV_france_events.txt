## ------------------------ Champagne Riots ------------------------ ##

## Champagne riots start
country_event = {
	id = wsv_france.2
	title = wsv_france.2.t
	desc = wsv_france.2.d

	fire_only_once = yes

    trigger = {
		original_tag = FRA
		has_war = no
		date > 1910.12.1   # historically started towards the end of 1910
		NOT = { has_global_flag = FRA_champagne_riots_occured }
	}

	mean_time_to_happen = { days = 25 }

	immediate = {
		set_global_flag = FRA_champagne_riots_occured
	}

	option = {
		name = wsv_france.2.a
		add_ideas = FRA_champagne_riots_idea
	}
}

## ----------------------- France Demands Alsace-Lorraine ------------------------ ##

# News of France demanding return of Alsace-Lorraine
news_event = {
	id = wsv_france.55
	title = wsv_france.55.t
	desc = wsv_france.55.d
	# picture = # TODO - picture

	is_triggered_only = yes
	major = yes

	immediate = {
		set_global_flag = france_demands_alsace_lorraine
		hidden_effect = {
			FRA = {
				add_named_threat = {
					threat = 10
					name = "France Demands the Return of Alsace-Lorraine"
				}
			}
			GER = { country_event = { id = wsv_france.56  days = 1 } }
			# mobilization flags (for ai strategies)
			every_country = {
				limit = { is_in_faction_with = FRA }
				set_country_flag = mobilize_against_centralpowers
			}
			every_country = {
				limit = { is_in_faction_with = GER }
				set_country_flag = mobilize_against_entente
			}
		}
	}

	option = {
		name = wsv_france.55.a
		trigger = { OR = { tag = FRA  is_in_faction_with = FRA } }
	}
	option = {
		name = wsv_france.55.b
		trigger = { OR = { tag = GER  is_in_faction_with = GER } }
	}
	option = {
		name = wsv_france.55.c
		trigger = { 
			NOT = { tag = FRA  is_in_faction_with = FRA } 
			NOT = { tag = GER  is_in_faction_with = GER } 
		}
	}
}
