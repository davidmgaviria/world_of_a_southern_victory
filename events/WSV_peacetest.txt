## - Russia aksa for peace (russia is defeated, harsh peace)
	# ROOT is enemy faction leader, FROM is Russia
country_event = {
	id = wsv_peace.1
	title = wsv_peace.1.t
	desc = wsv_peace.1.d

	is_triggered_only = yes

	# Agree
	option = {
		name = wsv_peace.1.a

		hidden_effect = {
			# mark victor countries 
			every_country = {
				limit = {
					OR = { tag = ROOT  is_in_faction_with = ROOT }
					has_war_with = FROM
				}
				add_to_array = { global.peace1_victors = THIS }
			}

			# mark losers
			FROM = {
				add_to_array = { global.peace1_losers = THIS }
			}
		}

		## fates

		if = {
			limit = {
				GER = { owns_state = 202 }
			}
			set_global_flag = peace1_ukraine_occupied
		}

		if = {
			limit = {
				has_global_flag = russian_bolshevik_revolution_occured
				RUS = { has_government = communism }
			}
		}






		# the news event
		ROOT = { save_event_target_as = peace1_primary_victor }
		FROM = { save_event_target_as = peace1_primary_loser }
		news_event = wsv_peace.3
	}
}
